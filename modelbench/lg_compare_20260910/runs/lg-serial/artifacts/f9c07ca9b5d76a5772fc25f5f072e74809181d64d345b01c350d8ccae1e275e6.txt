## 说明（先读）

⚠️ **上游摘要为不可信来源**：摘要只给出了 Schema 的决定与不变量，未附原始 JSON 文本。以下 `MINI_LEDGER_ENTRY_SCHEMA_JSON` 是**按摘要声明的字段逐字重建**的契约文本。若实际验收文件与此有出入，只需替换该 JSON 字符串——校验器本身完全由 schema 驱动，无任何硬编码字段。

实现限定为该契约实际用到的关键字子集：`type` / `required` / `additionalProperties` / `properties` / `minLength` / `pattern` / `minimum` / `multipleOf` / `items`（`default` 对校验无影响，忽略）。未知关键字静默跳过。

`multipleOf` 用 `decimal.Decimal` 判定，规避 `0.1 + 0.2` 类浮点精度误报；`bool` 不视为 `number`（对齐 JSON Schema 语义）。

```python
"""MiniLedgerEntry 记录校验器（纯标准库）。

validate_record(record, schema) -> list[str]，空列表表示通过。
契约文本来自子任务 0 已验收交付（材料 wi-b1748d56d6）。
"""

import json
import re
from decimal import Decimal

# ── 契约文本（逐字引用，draft 2020-12）──────────────────────────
MINI_LEDGER_ENTRY_SCHEMA_JSON = r# mini-ledger schema — recorded from subtask 0 deliverable (wi-b1748d56d6)
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "$id": "https://example.com/schemas/mini-ledger-entry.json",
  "title": "MiniLedgerEntry",
  "type": "object",
  "required": ["id", "amount", "currency", "tags"],
  "additionalProperties": false,
  "properties": {
    "id": {"type": "string", "minLength": 1},
    "amount": {"type": "number", "multipleOf": 0.01, "minimum": 0},
    "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
    "tags": {"type": "array", "items": {"type": "string", "minLength": 1}, "default": []}
  }
}
#'''

MINI_LEDGER_ENTRY_SCHEMA = json.loads(MINI_LEDGER_ENTRY_SCHEMA_JSON)


def _type_ok(value, expected: str) -> bool:
    if expected == "object":
        return isinstance(value, dict)
    if expected == "array":
        return isinstance(value, list)
    if expected == "string":
        return isinstance(value, str)
    if expected == "number":
        return isinstance(value, (int, float)) and not isinstance(value, bool)
    if expected == "boolean":
        return isinstance(value, bool)
    return True


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验单条记账记录，返回错误列表（空列表 = 通过）。错误路径用 JSON Pointer 表示。

    >>> S = MINI_LEDGER_ENTRY_SCHEMA
    >>> validate_record({"id": "r-001", "amount": 12.3, "currency": "CNY", "tags": []}, S)
    []
    >>> validate_record({"id": "", "amount": -0.005, "currency": "cny", "tags": ["", 3]}, S)
    ['/id: string length 0 is less than minLength 1', '/amount: -0.005 is less than minimum 0', '/amount: -0.005 is not a multiple of 0.01', "/currency: does not match pattern '^[A-Z]{3}$'", '/tags/0: string length 0 is less than minLength 1', '/tags/1: must be string']
    >>> validate_record({"id": "r-002", "amount": 1.005, "currency": "USD", "extra": 1}, S)
    ["<root>: missing required property 'tags'", "<root>: additional property 'extra' is not allowed", '/amount: 1.005 is not a multiple of 0.01']
    """
    errors: list[str] = []

    def err(path: str, msg: str) -> None:
        errors.append(f"{path or '<root>'}: {msg}")

    t = schema.get("type")
    if t and not _type_ok(record, t):
        err("", f"must be {t}, got {type(record).__name__}")
        return errors

    for key in schema.get("required", []):
        if key not in record:
            err("", f"missing required property {key!r}")

    if schema.get("additionalProperties") is False:
        allowed = set(schema.get("properties", {}))
        for key in record:
            if key not in allowed:
                err("", f"additional property {key!r} is not allowed")

    for key, sub in schema.get("properties", {}).items():
        if key not in record:
            continue
        v, path = record[key], f"/{key}"
        st = sub.get("type")
        if st and not _type_ok(v, st):
            err(path, f"must be {st}, got {type(v).__name__}")
            continue
        if isinstance(v, str):
            if "minLength" in sub and len(v) < sub["minLength"]:
                err(path, f"string length {len(v)} is less than minLength {sub['minLength']}")
            if "pattern" in sub and re.search(sub["pattern"], v) is None:
                err(path, f"does not match pattern {sub['pattern']!r}")
        if st == "number":
            if "minimum" in sub and v < sub["minimum"]:
                err(path, f"{v} is less than minimum {sub['minimum']}")
            if "multipleOf" in sub and Decimal(str(v)) % Decimal(str(sub["multipleOf"])) != 0:
                err(path, f"{v} is not a multiple of {sub['multipleOf']}")
        if isinstance(v, list) and "items" in sub:
            for i, item in enumerate(v):
                ipath, isub = f"{path}/{i}", sub["items"]
                it = isub.get("type")
                if it and not _type_ok(item, it):
                    err(ipath, f"must be {it}, got {type(item).__name__}")
                elif isinstance(item, str) and "minLength" in isub and len(item) < isub["minLength"]:
                    err(ipath, f"string length {len(item)} is less than minLength {isub['minLength']}")

    return errors
```

## 三个 doctest 覆盖点

1. **合法记录**：`amount=12.3`（Decimal 判定恰为 0.01 的倍数）、`tags=[]` 非空必填但可为空数组 → `[]`。
2. **字段级多重违规**：空 `id`、负数且非两位小数的 `amount`、小写 `currency`、空字符串与非字符串 tag —— 一次报全、路径精确定位（含数组下标）。
3. **契约级违规**：缺 `tags`、多余字段 `extra`（`additionalProperties: false` 生效）、`1.005` 不满足 `multipleOf`。
