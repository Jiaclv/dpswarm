```python
def validate_record(record: dict, schema: dict) -> list[str]:
    """校验单条 ledger 记录，返回违规信息列表；空列表表示通过。

    支持 type / required / additionalProperties / pattern / multipleOf /
    enum / maxLength / maxItems / items。pattern 为 JSON Schema 语义
    （re.search，非锚定）。multipleOf 用 Decimal 精确判定。

    >>> validate_record({"tx_id": "TX-000123", "amount": 19.99,
    ...                  "currency": "CNY"}, MINI_LEDGER_SCHEMA)
    []

    >>> validate_record({"tx_id": "TX-000123", "amount": 10.005,
    ...                  "currency": "USD"}, MINI_LEDGER_SCHEMA)
    ['amount: 10.005 violates multipleOf 0.01']

    >>> validate_record({"tx_id": "tx-9", "amount": 5.0, "currency": "GBP",
    ...                  "note": "extra"}, MINI_LEDGER_SCHEMA)
    ['tx_id: value violates pattern ^TX-[0-9]{6}$', 'currency: value not in enum', 'additionalProperties: unexpected key note']

    >>> validate_record({"tx_id": "TX-000001", "amount": 3.5},
    ...                 MINI_LEDGER_SCHEMA)
    ['required: missing key currency']
    """
    errors: list[str] = []
    _validate_object("", record, schema, errors)
    return errors


def _validate_object(path, obj, schema, errors):
    if schema.get("type") == "object":
        props = schema.get("properties", {})
        for key in schema.get("required", []):
            if key not in obj:
                errors.append(f"{_join(path, key)}: required key missing")
        if schema.get("additionalProperties") is False:
            for key in obj:
                if key not in props:
                    errors.append(f"{_join(path, key)}: unexpected key (additionalProperties=false)")
        for key, sub in props.items():
            if key in obj:
                _validate_value(_join(path, key), obj[key], sub, errors)


def _validate_value(path, value, rules, errors):
    t = rules.get("type")
    if t and not _check_type(value, t):
        errors.append(f"{path}: expected type {t}, got {type(value).__name__}")
        return
    if t == "string":
        pat = rules.get("pattern")
        if pat and not re.search(pat, value):
            errors.append(f"{path}: value violates pattern {pat}")
        if "maxLength" in rules and len(value) > rules["maxLength"]:
            errors.append(f"{path}: length {len(value)} exceeds maxLength {rules['maxLength']}")
    elif t == "number":
        m = rules.get("multipleOf")
        if m and Decimal(str(value)) % Decimal(str(m)) != 0:
            errors.append(f"{path}: {value} violates multipleOf {m}")
    if "enum" in rules and value not in rules["enum"]:
        errors.append(f"{path}: value not in enum {rules['enum']}")
    if t == "array":
        if "maxItems" in rules and len(value) > rules["maxItems"]:
            errors.append(f"{path}: {len(value)} items exceed maxItems {rules['maxItems']}")
        item_rules = rules.get("items")
        if item_rules:
            for i, item in enumerate(value):
                _validate_value(f"{path}[{i}]", item, item_rules, errors)


def _check_type(value, t):
    return {
        "string": lambda v: isinstance(v, str),
        "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
        "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
        "boolean": lambda v: isinstance(v, bool),
        "null": lambda v: v is None,
        "array": lambda v: isinstance(v, list),
        "object": lambda v: isinstance(v, dict),
    }[t](value)


def _join(path, key):
    return key if not path else f"{path}.{key}"


if __name__ == "__main__":
    import doctest
    result = doctest.testmod(verbose=False)
    print(f"doctest: {result.attempted} cases, {result.failed} failed")
```
