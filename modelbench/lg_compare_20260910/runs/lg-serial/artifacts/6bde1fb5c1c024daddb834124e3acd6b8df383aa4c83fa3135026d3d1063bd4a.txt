import re

# ==== 以下为 A 交付文本 wi-3fb713fa50 的逐字引用，禁止改动 ====
_A_SCHEMA_TEXT = """\
{
  "type": "object",
  "required": ["id", "name", "tags"],
  "properties": {

SCHEMA = json.loads(_A_SCHEMA_TEXT)

_TYPE_NAMES = {
    str: "string", bool: "boolean", int: "integer", float: "number",
    list: "array", dict: "object", type(None): "null",
}


def _type_name(value):
    return _TYPE_NAMES[type(value)]


def _check_type(value, expected):
    if expected == "string":
        return isinstance(value, str)
    if expected == "integer":
        return isinstance(value, int) and not isinstance(value, bool)
    if expected == "number":
        return isinstance(value, (int, float)) and not isinstance(value, bool)
    if expected == "boolean":
        return isinstance(value, bool)
    if expected == "array":
        return isinstance(value, list)
    if expected == "object":
        return isinstance(value, dict)
    if expected == "null":
        return value is None
    return True


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表 = 通过）。

    用例中的 schema 均逐字引用 A 的交付文本（见 _A_SCHEMA_TEXT / SCHEMA）。

    >>> validate_record({"id": "XY-0001", "name": "Alice", "tags": ["alpha"], "age": 30}, SCHEMA)
    []

    >>> validate_record({"id": "abc", "tags": []}, SCHEMA)
    ['name: required property is missing', 'id: value does not match pattern ^[A-Z]{2}-[0-9]{4}$', 'tags: array length 0 is less than minItems 1']

    >>> validate_record({"id": "AB-1234", "name": "Bob", "tags": ["x", "x"], "age": 200, "role": "admin"}, SCHEMA)
    ['tags: items are not unique', 'age: value 200 is above maximum 150', 'role: additional property is not allowed']
    """
    if not isinstance(record, dict):
        return ["record: expected object, got " + _type_name(record)]

    errors: list[str] = []
    props = schema.get("properties", {})

    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"{key}: required property is missing")

    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in props:
                errors.append(f"{key}: additional property is not allowed")

    for key, sub in props.items():
        if key not in record:
            continue
        value = record[key]
        expected = sub.get("type")
        if expected and not _check_type(value, expected):
            errors.append(f"{key}: expected {expected}, got {_type_name(value)}")
            continue
        if expected == "string":
            if "minLength" in sub and len(value) < sub["minLength"]:
                errors.append(f"{key}: string length {len(value)} is less than minLength {sub['minLength']}")
            if "maxLength" in sub and len(value) > sub["maxLength"]:
                errors.append(f"{key}: string length {len(value)} is above maxLength {sub['maxLength']}")
        elif expected == "integer" or expected == "number":
            if "minimum" in sub and value < sub["minimum"]:
                errors.append(f"{key}: value {value} is below minimum {sub['minimum']}")
            if "maximum" in sub and value > sub["maximum"]:
                errors.append(f"{key}: value {value} is above maximum {sub['maximum']}")
        elif expected == "array":
            if "minItems" in sub and len(value) < sub["minItems"]:
                errors.append(f"{key}: array length {len(value)} is less than minItems {sub['minItems']}")
            if "maxItems" in sub and len(value) > sub["maxItems"]:
                errors.append(f"{key}: array length {len(value)} is above maxItems {sub['maxItems']}")
            if sub.get("uniqueItems") and len({repr(item) for item in value}) != len(value):
                errors.append(f"{key}: items are not unique")
            item_schema = sub.get("items")
            if item_schema:
                item_type = item_schema.get("type")
                for i, item in enumerate(value):
                    if item_type and not _check_type(item, item_type):
                        errors.append(f"{key}[{i}]: expected {item_type}, got {_type_name(item)}")

    return errors
