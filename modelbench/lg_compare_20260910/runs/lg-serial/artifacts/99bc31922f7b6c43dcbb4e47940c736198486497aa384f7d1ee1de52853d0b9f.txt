schema 逐字引用 A 验收后的交付（pkg://wi-700747abd5-r1/9a292c3fab0b）。"""

# ---- A 交付原文 · SCHEMA（逐字复制，未做任何改动）----
SCHEMA = {
    "type": "object",
    "required": ["id", "name", "tags"],
    "properties": {
        "id":   {"type": "integer"},
        "name": {"type": "string"},
        "tags": {"type": "array", "items": {"type": "string"}},
    },
}
# ---- A 交付原文结束 ----

_TYPE_CHECKS = {
    "object":  lambda v: isinstance(v, dict),
    "array":   lambda v: isinstance(v, list),
    "string":  lambda v: isinstance(v, str),
    "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
    "number":  lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
    "boolean": lambda v: isinstance(v, bool),
    "null":    lambda v: v is None,
}


def validate_record(record: dict, schema: dict) -> list[str]:
    """按 schema 校验 record，返回错误列表；空列表表示通过。

    仅使用内建能力；支持 schema 关键字 type / required / properties /
    items（items 递归）。错误消息为点路径风格，如
    "name: missing required property"。

    >>> validate_record({"id": 1, "name": "nt", "tags": ["a"]}, SCHEMA)
    []
    >>> validate_record({"id": "1", "name": "nt", "tags": ["a"]}, SCHEMA)
    ['id: expected integer, got str']
    >>> validate_record({"id": 2, "tags": ["a", 3]}, SCHEMA)
    ['name: missing required property', 'tags[1]: expected string, got int']
    """
    return _validate(record, schema, "")


def _validate(value, schema: dict, path: str) -> list[str]:
    errors: list[str] = []
    expected = schema.get("type")
    if expected is not None:
        check = _TYPE_CHECKS.get(expected)
        if check is None:
            errors.append(f"{path or '<root>'}: unknown type {expected!r}")
            return errors
        if not check(value):
            errors.append(
                f"{path or '<root>'}: expected {expected}, got {type(value).__name__}"
            )
            return errors
    if isinstance(value, dict):
        for key in schema.get("required", ()):
            if key not in value:
                errors.append(
                    f"{path + '.' + key if path else key}: missing required property"
                )
        for key, sub in schema.get("properties", {}).items():
            if key in value:
                errors.extend(
                    _validate(value[key], sub, f"{path}.{key}" if path else key)
                )
    if isinstance(value, list):
        items = schema.get("items")
        if isinstance(items, dict):
            for i, item in enumerate(value):
                errors.extend(_validate(item, items, f"{path}[{i}]"))
    return errors


if __name__ == "__main__":
