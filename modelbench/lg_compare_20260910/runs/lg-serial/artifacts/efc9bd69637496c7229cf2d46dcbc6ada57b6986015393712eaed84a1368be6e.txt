def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表 = 通过）。

    支持：required 必填检查、type 类型检查（integer/string/number/boolean）、
    minimum/maximum 数值范围、minLength 字符串最短长度。

    >>> SCHEMA_FROM_A = {
    ...     "type": "object",
    ...     "required": ["id", "name", "age"],
    ...     "properties": {
    ...         "id": {"type": "integer"},
    ...         "name": {"type": "string", "minLength": 1},
    ...         "age": {"type": "integer", "minimum": 0, "maximum": 150},
    ...     },
    ... }
    >>> validate_record({"id": 1, "name": "Alice", "age": 30}, SCHEMA_FROM_A)
    []
    >>> validate_record({"id": "x", "age": -1}, SCHEMA_FROM_A)
    ["缺少必填字段: 'name'", "字段 'id' 类型错误: 期望 integer, 实际 str", "字段 'age' 违反 minimum=0"]
    >>> validate_record({"id": 2, "name": "", "age": 999}, SCHEMA_FROM_A)
    ["字段 'name' 长度 0 违反 minLength=1", "字段 'age' 违反 maximum=150"]
    """
    violations: list[str] = []

    props = schema.get("properties", {})
    for key in schema.get("required", []):
        if key not in record:
            violations.append(f"缺少必填字段: {key!r}")

    for key, value in record.items():
        spec = props.get(key)
        if spec is None:
            continue
        expected = spec.get("type")
        if expected == "integer" and not isinstance(value, int):
            violations.append(f"字段 {key!r} 类型错误: 期望 integer, 实际 {type(value).__name__}")
        elif expected == "string" and not isinstance(value, str):
            violations.append(f"字段 {key!r} 类型错误: 期望 string, 实际 {type(value).__name__}")
        elif expected == "number" and not isinstance(value, (int, float)):
            violations.append(f"字段 {key!r} 类型错误: 期望 number, 实际 {type(value).__name__}")
        elif expected == "boolean" and not isinstance(value, bool):
            violations.append(f"字段 {key!r} 类型错误: 期望 boolean, 实际 {type(value).__name__}")

        if isinstance(value, (int, float)) and not isinstance(value, bool):
            if "minimum" in spec and value < spec["minimum"]:
                violations.append(f"字段 {key!r} 违反 minimum={spec['minimum']}")
            if "maximum" in spec and value > spec["maximum"]:
                violations.append(f"字段 {key!r} 违反 maximum={spec['maximum']}")
        if isinstance(value, str) and "minLength" in spec and len(value) < spec["minLength"]:
            violations.append(f"字段 {key!r} 长度 {len(value)} 违反 minLength={spec['minLength']}")

    return violations
