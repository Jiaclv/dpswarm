def to_roman(n: int) -> str:
    """将 1<=n<=3999 的整数转换为罗马数字字符串。

    >>> to_roman(1)
    'I'
    >>> to_roman(1994)
    'MCMXCIV'
    >>> to_roman(3999)
    'MMMCMXCIX'
    """
    vals = [(1000, 'M'), (900, 'CM'), (500, 'D'), (400, 'CD'),
            (100, 'C'), (90, 'XC'), (50, 'L'), (40, 'XL'),
            (10, 'X'), (9, 'IX'), (5, 'V'), (4, 'IV'), (1, 'I')]
    result = []
    for v, s in vals:
        while n >= v:
            result.append(s)
            n -= v
    return ''.join(result)
