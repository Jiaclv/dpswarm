## 实现（纯标准库）

```python
def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回错误列表（空列表 = 通过）。

    支持的 schema 关键字：type / required / properties /
    items / minLength / maxLength / minimum / maximum / enum。

    >>> SCHEMA = PLACEHOLDER_SCHEMA  # TODO: 逐字替换为 A 交付的 schema 文本
    >>> validate_record({"id": 1, "name": "alice", "tags": ["x"]}, SCHEMA)
    []
    >>> validate_record({"id": "1", "name": ""}, SCHEMA)
    ["id: 期望 integer, 实际 str", "name: 长度 0 < minLength 1"]
    >>> validate_record({"name": "bob"}, SCHEMA)
    ["缺少必填字段: id"]
    """
    errors: list[str] = []

    expected = schema.get("type")
    if expected and not _type_ok(record, expected):
        errors.append(f"根对象: 期望 {expected}, 实际 {type(record).__name__}")
        return errors

    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"缺少必填字段: {key}")

    for key, rule in schema.get("properties", {}).items():
        if key not in record:
            continue
        errors.extend(_validate_value(key, record[key], rule))

    return errors


def _type_ok(value, expected) -> bool:
    py_types = {
        "object": dict, "array": list, "string": str,
        "integer": int, "number": (int, float), "boolean": bool, "null": type(None),
    }
    t = py_types.get(expected)
    if t is None:
        return True
    # bool 是 int 的子类，需排除
    if expected == "integer" and isinstance(value, bool):
        return False
    return isinstance(value, t)


def _validate_value(key: str, value, rule: dict) -> list[str]:
    errors: list[str] = []
    expected = rule.get("type")
    if expected and not _type_ok(value, expected):
        errors.append(f"{key}: 期望 {expected}, 实际 {type(value).__name__}")
        return errors

    if "enum" in rule and value not in rule["enum"]:
        errors.append(f"{key}: {value!r} 不在 enum {rule['enum']!r} 中")

    if isinstance(value, str):
        if "minLength" in rule and len(value) < rule["minLength"]:
            errors.append(f"{key}: 长度 {len(value)} < minLength {rule['minLength']}")
        if "maxLength" in rule and len(value) > rule["maxLength"]:
            errors.append(f"{key}: 长度 {len(value)} > maxLength {rule['maxLength']}")

    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in rule and value < rule["minimum"]:
            errors.append(f"{key}: {value} < minimum {rule['minimum']}")
        if "maximum" in rule and value > rule["maximum"]:
            errors.append(f"{key}: {value} > maximum {rule['maximum']}")

    if isinstance(value, list) and "items" in rule:
        for i, item in enumerate(value):
            errors.extend(_validate_value(f"{key}[{i}]", item, rule["items"]))

    return errors
```
