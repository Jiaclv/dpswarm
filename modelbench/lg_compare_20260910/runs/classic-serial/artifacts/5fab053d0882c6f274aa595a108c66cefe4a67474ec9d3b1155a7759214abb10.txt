"""record 校验器：纯标准库实现，依据 A 验收后的 schema 文本。

schema 支持关键字：type / properties / required / enum / maxLength /
minimum / maximum。validate_record 返回错误信息列表，空列表表示通过。
"""

# 以下 SCHEMA 逐字引用自 pkg://wi-3f915d6080-r1/911a2317e2c5（A 验收后交付）
SCHEMA = {
    "type": "object",
    "properties": {
        "id": {"type": "integer", "minimum": 1},
        "name": {"type": "string", "maxLength": 20},
        "role": {"type": "string", "enum": ["admin", "user", "guest"]},
        "score": {"type": "number", "minimum": 0, "maximum": 100},
    },
    "required": ["id", "name"],
}


def _type_ok(value, t):
    """判断 value 是否符合 schema 的 type 关键字。bool 不算 integer/number。"""
    if t == "object":
        return isinstance(value, dict)
    if t == "array":
        return isinstance(value, list)
    if t == "string":
        return isinstance(value, str)
    if t == "integer":
        return isinstance(value, int) and not isinstance(value, bool)
    if t == "number":
        return isinstance(value, (int, float)) and not isinstance(value, bool)
    if t == "boolean":
        return isinstance(value, bool)
    if t == "null":
        return value is None
    return False  # 未知 type：不通过，避免静默放行


def _validate_value(path, value, spec, errors):
    """对单个字段按其 schema 片段校验，错误追加到 errors。"""
    t = spec.get("type")
    if t is not None and not _type_ok(value, t):
        errors.append(
            f"{path}: type mismatch, expected {t}, got {type(value).__name__}"
        )
        return  # 类型不符时不再做值域检查，避免对异型值误报
    if "enum" in spec and value not in spec["enum"]:
        errors.append(f"{path}: value {value!r} not in enum {spec['enum']}")
    if isinstance(value, str) and "maxLength" in spec and len(value) > spec["maxLength"]:
        errors.append(f"{path}: length {len(value)} exceeds maxLength {spec['maxLength']}")
    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in spec and value < spec["minimum"]:
            errors.append(f"{path}: value {value} below minimum {spec['minimum']}")
        if "maximum" in spec and value > spec["maximum"]:
            errors.append(f"{path}: value {value} above maximum {spec['maximum']}")


def validate_record(record, schema):
    """校验单条记录，返回错误信息列表（空列表 = 通过）。

    >>> validate_record({"id": 1, "name": "alice", "role": "user", "score": 88}, SCHEMA)
    []
    >>> validate_record({"id": "x", "score": 5}, SCHEMA)
    ['name: required field missing', 'id: type mismatch, expected integer, got str']
    >>> validate_record({"id": 0, "name": "n" * 21, "role": "root", "score": 101}, SCHEMA)
    ['id: value 0 below minimum 1', 'name: length 21 exceeds maxLength 20', 'role: value 'root' not in enum ['admin', 'user', 'guest']', 'score: value 101 above maximum 100']
    """
    errors = []
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"{key}: required field missing")
    props = schema.get("properties", {})
    for key, value in record.items():  # dict 保持插入序，输出可复现
        if key in props:
            _validate_value(key, value, props[key], errors)
    return errors
