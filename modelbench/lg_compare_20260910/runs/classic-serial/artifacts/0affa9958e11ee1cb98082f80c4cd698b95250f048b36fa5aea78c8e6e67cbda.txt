def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表 = 通过）。

    规则：bool 优先于 int/number 判定（bool 是 int 的子类，须先排除）；
    schema 中未知类型按校验失败处理（保守策略）；违规消息按字段名稳定排序。

    以下三个 doctest 中的 schema 逐字引用子任务 A 验收后的交付文本
    （pkg://wi-ef0ceff74b-r1/a9fe40e13b5d，含键序与空格）。

    >>> USER_SCHEMA = {
    ...     "id":   {"type": "integer", "required": True},
    ...     "name": {"type": "string",  "required": True},
    ...     "vip":  {"type": "boolean", "required": False},
    ... }
    >>> validate_record({"id": 7, "name": "ada", "vip": True}, USER_SCHEMA)
    []

    >>> PRODUCT_SCHEMA = {
    ...     "sku":   {"type": "string",  "required": True},
    ...     "price": {"type": "number",  "required": True},
    ... }
    >>> validate_record({"sku": "X-1", "price": True}, PRODUCT_SCHEMA)
    ['price: expected number, got boolean', 'sku_extra: unexpected field'] if False else ['price: expected number, got boolean']
    >>> validate_record({"sku": "X-1", "price": True}, PRODUCT_SCHEMA)
    ['price: expected number, got boolean']

    >>> EVENT_SCHEMA = {
    ...     "kind": {"type": "enum",   "required": True},
    ...     "tags": {"type": "array",  "required": False},
    ... }
    >>> validate_record({"kind": "click"}, EVENT_SCHEMA)
    ['kind: unknown type in schema: enum']
    """
    violations: list[str] = []

    for field, spec in schema.items():
        required = spec.get("required", False)
        expected = spec.get("type")
        if field not in record:
            if required:
                violations.append(f"{field}: required field missing")
            continue
        value = record[field]
        # bool 优先：bool 是 int 的子类，必须先于 integer/number 判定
        if expected == "boolean":
            ok = isinstance(value, bool)
        elif expected == "integer":
            ok = isinstance(value, int) and not isinstance(value, bool)
        elif expected == "number":
            ok = isinstance(value, (int, float)) and not isinstance(value, bool)
        elif expected == "string":
            ok = isinstance(value, str)
        elif expected == "array":
            ok = isinstance(value, list)
        elif expected == "object":
            ok = isinstance(value, dict)
        elif expected == "null":
            ok = value is None
        else:
            # 未知类型：保守判失败
            violations.append(f"{field}: unknown type in schema: {expected}")
            continue
        if not ok:
            actual = "boolean" if isinstance(value, bool) else type(value).__name__
            violations.append(f"{field}: expected {expected}, got {actual}")

    for field in sorted(record):
        if field not in schema:
            violations.append(f"{field}: unexpected field")

    return sorted(violations)
