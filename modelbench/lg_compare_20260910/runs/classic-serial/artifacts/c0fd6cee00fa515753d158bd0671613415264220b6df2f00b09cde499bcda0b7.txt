"""record 校验器 —— 纯标准库实现。

import re
from decimal import Decimal, InvalidOperation

# ── subtask 0 交付的 schema 全文（逐字引用，未做任何改动） ──────────────
SCHEMA = {
    "type": "object",
    "required": ["order_id", "amount", "currency", "tags"],
    "properties": {
        "order_id": {"type": "string"},
        "amount": {"type": "number", "format": "decimal(2)"},
        "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
        "tags": {"type": "array", "nullable": True}
    }
}
# ──────────────────────────────────────────────────────────────────────


def _type_ok(value, type_name):
    """按 schema 的 type 关键字判定类型。bool 不计入 number/string。"""
    if type_name == "string":
        return isinstance(value, str)
    if type_name == "number":
        return isinstance(value, (int, float, Decimal)) and not isinstance(value, bool)
    if type_name == "array":
        return isinstance(value, list)
    return False


def _amount_at_most_2dp(value):
    """amount 须满足 format 'decimal(2)'：至多两位小数。

    用 Decimal 的 exponent 判定，避免 0.1+0.2 类浮点表示误差；
    非有限值（inf/nan）一律拒绝。
    """
    if isinstance(value, bool) or not isinstance(value, (int, float, Decimal)):
        return False
    try:
        d = Decimal(str(value))
    except InvalidOperation:
        return False
    if not d.is_finite():
        return False
    return d.as_tuple().exponent >= -2


def validate_record(record, schema):
    """校验 record 是否满足 schema，返回违规信息列表（空列表 = 通过）。

    覆盖约束：required / 各字段 type / amount 至多两位小数 /
    currency 三字母大写 / tags 数组可空。

    >>> validate_record(
    ...     {"order_id": "A-001", "amount": 12.5, "currency": "USD", "tags": []},
    ...     SCHEMA,
    ... )
    []

    >>> validate_record(
    ...     {"order_id": 7, "amount": "x", "currency": "usd1", "tags": "a b"},
    ...     SCHEMA,
    ... )
    ['order_id: expected string', 'amount: expected number', 'amount: expected at most 2 decimal places', 'currency: expected 3 uppercase letters', 'tags: expected array or null']

    >>> validate_record(
    ...     {"order_id": "A-002", "amount": 1.999, "currency": "JPY"},
    ...     SCHEMA,
    ... )
    ['amount: expected at most 2 decimal places', 'missing required field: tags']

    解释性说明：'两位小数'按 schema 的 format 'decimal(2)' 解释为
    **至多**两位小数（整数与一位小数均合法）。
    """
    violations = []

    # required
    for field in schema.get("required", []):
        if field not in record:
            violations.append(f"missing required field: {field}")

    for field, rule in schema.get("properties", {}).items():
        if field not in record:
            continue
        value = record[field]

        # nullable（tags 数组可空）
        if rule.get("nullable") and value is None:
            continue

        # type
        expected = rule.get("type")
        if expected and not _type_ok(value, expected):
            violations.append(f"{field}: expected {expected}")
            continue

        # amount: format decimal(2)
        if rule.get("format") == "decimal(2)":
            if not _amount_at_most_2dp(value):
                violations.append(f"{field}: expected at most 2 decimal places")

        # currency: pattern ^[A-Z]{3}$
        if "pattern" in rule:
            if not isinstance(value, str) or not re.fullmatch(rule["pattern"], value):
                violations.append(f"{field}: expected 3 uppercase letters")

    return violations
