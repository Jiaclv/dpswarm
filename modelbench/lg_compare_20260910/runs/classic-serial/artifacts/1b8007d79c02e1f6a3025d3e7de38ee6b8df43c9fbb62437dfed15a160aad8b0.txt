## 交付：validate_record 实现 + 3 个 doctest

```python
"""子任务 B：校验器实现（纯标准库，纯函数）。

错误格式（与 Lead 验收约定一致）：
    "<字段路径>: <错误描述>"，路径规则：顶层字段直接用字段名，
    无嵌套对象支持（A 的 schema 仅一层 properties）。
空列表 [] = 校验通过。

schema 为 A 验收后的原文，逐字引用于 SCHEMA 常量，不得改写。
"""

import re

# ---- A 验收后的 schema（逐字引用，勿改） ----
SCHEMA = {
    "draft": "dp-swarm/1",
    "type": "object",
    "required": ["id", "email"],
    "properties": {
        "id": {"type": "string", "pattern": "^[a-z0-9-]+$"},
        "email": {"type": "string", "format": "email"},
        "age": {"type": "integer", "minimum": 0, "maximum": 150},
    },
}

_EMAIL_RE = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")
_TYPE_NAMES = {"string": str, "integer": int}


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回错误列表（空列表=通过）。

    >>> validate_record({"id": "user-1", "email": "a@b.com", "age": 30}, SCHEMA)
    []

    >>> validate_record({"id": "User_1!", "email": "a@b.com", "age": "x"}, SCHEMA)
    ['id: value does not match pattern ^[a-z0-9-]+$', 'age: expected integer, got str']

    >>> validate_record({"email": "not-an-email", "age": 200}, SCHEMA)
    ['id: required field missing', 'email: invalid email format', 'age: value 200 exceeds maximum 150']
    """
    errors: list[str] = []

    # required 检查（按 schema 声明顺序）
    for field in schema.get("required", []):
        if field not in record:
            errors.append(f"{field}: required field missing")

    # properties 检查
    props = schema.get("properties", {})
    for field, rule in props.items():
        if field not in record:
            continue
        value = record[field]

        expected = _TYPE_NAMES.get(rule.get("type"))
        if expected is not None and not isinstance(value, expected):
            errors.append(f"{field}: expected {rule['type']}, got {type(value).__name__}")
            continue

        if "pattern" in rule and isinstance(value, str):
            if not re.fullmatch(rule["pattern"], value):
                errors.append(f"{field}: value does not match pattern {rule['pattern']}")

        if rule.get("format") == "email" and isinstance(value, str):
            if not _EMAIL_RE.fullmatch(value):
                errors.append(f"{field}: invalid email format")

        if isinstance(value, int) and not isinstance(value, bool):
            if "minimum" in rule and value < rule["minimum"]:
                errors.append(f"{field}: value {value} below minimum {rule['minimum']}")
            if "maximum" in rule and value > rule["maximum"]:
                errors.append(f"{field}: value {value} exceeds maximum {rule['maximum']}")

    return errors


if __name__ == "__main__":
    import doctest
    doctest.testmod(verbose=False)
```
