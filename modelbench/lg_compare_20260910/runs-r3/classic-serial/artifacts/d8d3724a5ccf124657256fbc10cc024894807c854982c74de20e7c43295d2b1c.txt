def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规列表（空列表=通过）。

    违规排列顺序：先 required 缺失（按 schema['required'] 声明顺序），
    后逐 property（按 schema['properties'] 键序）。文档声明与实现一致。
    """
    violations: list[str] = []
    # ① required 缺失，按声明顺序
    for name in schema.get("required", []):
        if name not in record:
            violations.append(f"missing required field: {name}")
    # ② 逐 property，按 schema['properties'] 键序
    for name, prop in schema.get("properties", {}).items():
        if name not in record:
            continue
        value = record[name]
        if "type" in prop:
            checks = {
                "string": lambda v: isinstance(v, str),
                "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
                "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
                "boolean": lambda v: isinstance(v, bool),
            }
            if not checks.get(prop["type"], lambda v: True)(value):
                violations.append(f"{name}: expected type {prop['type']}")
                continue
        if isinstance(value, (int, float)) and not isinstance(value, bool) \
                and "minimum" in prop and value < prop["minimum"]:
            violations.append(f"{name}: {value} < minimum {prop['minimum']}")
        if isinstance(value, str) and "maxLength" in prop \
                and len(value) > prop["maxLength"]:
            violations.append(
                f"{name}: length {len(value)} > maxLength {prop['maxLength']}")
    return violations
