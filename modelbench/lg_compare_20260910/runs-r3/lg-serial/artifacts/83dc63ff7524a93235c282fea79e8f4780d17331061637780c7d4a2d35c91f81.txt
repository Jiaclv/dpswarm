# 子任务B：`validate_record` 实现

## 实现说明（决策先行）

- **覆盖范围**：仅实现 schema 中出现的 7 类关键字 —— `type`、`properties`、`required`、`items`、`pattern`、`multipleOf`、`additionalProperties: false`。纯标准库（`re` / `json` / `decimal` / `doctest`），零三方依赖。
- **浮点陷阱**：`multipleOf: 0.01` 直接用浮点除会因二进制精度误判（如 `1.005`）。经 `Decimal(str(value))` 走十进制字符串通道取余，`1.005 → 余 0.005 ≠ 0` 判违，`12.34 → 余 0` 判过。
- **JSON 类型语义**：`number` 接受 `int`/`float` 但排除 `bool`（JSON 中 `true` 是 boolean 非 number）；`pattern` 按 JSON Schema 规范用 `re.search`（非锚定语义，本题 pattern 自带 `^...$` 锚定，行为一致）。
- **错误顺序确定性**：`required` → `additionalProperties` → 按 schema 声明顺序逐属性 → 数组元素按下标。dict 插入序（Python 3.7+）保证 doctest 输出可复现。
- **逐字引用**：schema 以 `SCHEMA_TEXT` 原始字符串逐字嵌入（与 L1 层 [wi-c8db6ef2ef] 完全一致，含缩进与换行），经 `json.loads` 加载为 `SCHEMA` 供 doctest 引用，杜绝转抄失真。

## 代码

```python
# validator.py — 迷你记账记录校验器（子任务B）
# 依据：子任务A验收交付 [wi-c8db6ef2ef]，$schema draft 2020-12 子集实现。
from __future__ import annotations

import json
import re
from decimal import Decimal, InvalidOperation

# ---- 子任务A交付文本逐字引用（L1 层 [wi-c8db6ef2ef]，勿改动任何字符）----
SCHEMA_TEXT = """\
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "$id": "https://example.com/schemas/mini-ledger-entry.json",
  "title": "迷你记账记录",
  "type": "object",
  "properties": {
    "id": { "type": "string" },
    "amount": {
      "type": "number",
      "multipleOf": 0.01
    },
    "currency": {
      "type": "string",
      "pattern": "^[A-Z]{3}$"
    },
    "tags": {
      "type": ["array", "null"],
      "items": { "type": "string" }
    }
  },
  "required": ["id", "amount", "currency", "tags"],
  "additionalProperties": false
}
"""
SCHEMA = json.loads(SCHEMA_TEXT)


def _json_type(value: object) -> str:
    """Python 值 → JSON 类型名（bool 优先于 number 判定）。"""
    if value is None:
        return "null"
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, str):
        return "string"
    if isinstance(value, list):
        return "array"
    if isinstance(value, dict):
        return "object"
    return type(value).__name__


def _type_ok(value: object, allowed: list[str]) -> bool:
    actual = _json_type(value)
    if actual == "number" and "number" in allowed:
        return True
    return actual in allowed


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验单条记账记录，返回违规信息列表（空列表 = 通过）。

    覆盖关键字：type / properties / required / items / pattern /
    multipleOf / additionalProperties=false。

    >>> validate_record({"id": "a1", "amount": 12.34, "currency": "USD", "tags": ["food", "travel"]}, SCHEMA)
    []

    >>> validate_record({"id": "a2", "amount": 1.005, "currency": "usd", "tags": None}, SCHEMA)
    ['amount: not a multiple of 0.01', "currency: expected match to pattern '^[A-Z]{3}$'"]

    >>> validate_record({"id": "a3", "amount": 5, "currency": "JPY", "tags": ["x", 1], "memo": "extra"}, SCHEMA)
    ["additionalProperties: unexpected property 'memo'", 'tags[1]: expected type string, got number']
    """
    errors: list[str] = []

    if not isinstance(record, dict):
        return [f"root: expected type object, got {_json_type(record)}"]

    # 1) required
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"required: missing property '{key}'")

    # 2) additionalProperties: false
    props: dict = schema.get("properties", {})
    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in props:
                errors.append(f"additionalProperties: unexpected property '{key}'")

    # 3) 逐属性校验（按 schema 声明顺序，保证输出确定性）
    for key, sub in props.items():
        if key not in record:
            continue
        value = record[key]

        raw_type = sub.get("type")
        allowed = [raw_type] if isinstance(raw_type, str) else list(raw_type or [])
        if allowed and not _type_ok(value, allowed):
            errors.append(
                f"{key}: expected type {' or '.join(allowed)}, got {_json_type(value)}"
            )
            continue

        if isinstance(value, str) and "pattern" in sub:
            try:
                if re.search(sub["pattern"], value) is None:
                    errors.append(f"{key}: expected match to pattern {sub['pattern']!r}")
            except re.error as exc:
                errors.append(f"{key}: invalid pattern {sub['pattern']!r} ({exc})")

        if isinstance(value, (int, float)) and not isinstance(value, bool) and "multipleOf" in sub:
            try:
                if Decimal(str(value)) % Decimal(str(sub["multipleOf"])) != 0:
                    errors.append(f"{key}: not a multiple of {sub['multipleOf']}")
            except InvalidOperation:
                errors.append(f"{key}: not a multiple of {sub['multipleOf']}")

        if isinstance(value, list) and isinstance(sub.get("items"), dict):
            item_sub = sub["items"]
            for i, item in enumerate(value):
                if not _type_ok(item, [item_sub["type"]]):
                    errors.append(
                        f"{key}[{i}]: expected type {item_sub['type']}, got {_json_type(item)}"
                    )

    return errors


if __name__ == "__main__":
    import doctest
    raise SystemExit(1 if doctest.testmod(verbose=False).failed else 0)
```
