## 最终交付（可运行）

```python
"""MiniLedgerEntry 校验器 —— 纯标准库实现。

schema 逐字引用自子任务 0 已验收交付 [wi-d2bfed460e]。
"""

import json
import re
from decimal import Decimal


# ===== 子任务 0 交付，逐字引用开始 [wi-d2bfed460e] =====
_SCHEMA_TEXT = r'''
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "MiniLedgerEntry",
  "description": "迷你记账记录",
  "type": "object",
  "required": ["id", "amount", "currency", "tags"],
  "properties": {
    "id": {
      "type": "string",
      "minLength": 1
    },
    "amount": {
      "type": "number",
      "multipleOf": 0.01
    },
    "currency": {
      "type": "string",
      "pattern": "^[A-Z]{3}$",
      "minLength": 3,
      "maxLength": 3
    },
    "tags": {
      "type": ["array", "null"],
      "items": { "type": "string" },
      "uniqueItems": true
    }
  },
  "additionalProperties": false
}
'''
# ===== 子任务 0 交付，逐字引用结束 =====

SCHEMA = json.loads(_SCHEMA_TEXT)


def _check_type(value, type_spec) -> bool:
    """检查值是否符合 type 声明（支持字符串或字符串列表）。"""
    types = type_spec if isinstance(type_spec, list) else [type_spec]
    for t in types:
        if t == "object" and isinstance(value, dict):
            return True
        if t == "array" and isinstance(value, list):
            return True
        if t == "string" and isinstance(value, str):
            return True
        if t == "null" and value is None:
            return True
        if t == "number" and isinstance(value, (int, float)) and not isinstance(value, bool):
            return True
        if t == "integer" and isinstance(value, int) and not isinstance(value, bool):
            return True
        if t == "boolean" and isinstance(value, bool):
            return True
    return False


def _check_multiple_of(value, divisor) -> bool:
    """用 Decimal 精确判定 multipleOf，规避浮点误差。"""
    return Decimal(str(value)) % Decimal(str(divisor)) == 0


def validate_record(record: dict, schema: dict = SCHEMA) -> list[str]:
    """校验记账记录，返回违规信息列表（空列表 = 通过）。

    依据 schema：MiniLedgerEntry（draft-07，[wi-d2bfed460e]）。

    >>> valid = {"id": "tx-001", "amount": 19.99,
    ...          "currency": "CNY", "tags": ["food", None]}
    >>> validate_record(valid)
    []

    >>> bad = {"id": "", "amount": 0.1,
    ...        "currency": "cny", "tags": ["a", "a"], "extra": 1}
    >>> for msg in validate_record(bad): msg
    'id: minLength 违规（长度 0 < 1）'
    'amount: multipleOf 违规（0.1 不是 0.01 的整数倍）'
    'currency: pattern 违规（不匹配 ^[A-Z]{3}$）'
    'tags: uniqueItems 违规（存在重复元素）'
    'additionalProperties: 不允许的字段 "extra"'

    >>> missing = {"id": "tx-002", "amount": 5}
    >>> validate_record(missing)
    ['required: 缺少必填字段 currency', 'required: 缺少必填字段 tags']
    """
    errors: list[str] = []

    # 顶层：type=object
    if not isinstance(record, dict):
        return [f"type: 期望 object，实际 {type(record).__name__}"]

    # required
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"required: 缺少必填字段 {key}")

    # additionalProperties: false
    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in schema.get("properties", {}):
                errors.append(f'additionalProperties: 不允许的字段 "{key}"')

    # properties
    for key, subschema in schema.get("properties", {}).items():
        if key not in record:
            continue
        value = record[key]

        # type（支持 ["array", "null"] 联合类型）
        if "type" in subschema and not _check_type(value, subschema["type"]):
            expected = subschema["type"]
            expected = "/".join(expected) if isinstance(expected, list) else expected
            errors.append(f"{key}: type 违规（期望 {expected}，"
                          f"实际 {type(value).__name__}）")
            continue  # 类型不符时跳过后续断言，避免级联报错

        # minLength / maxLength
        if isinstance(value, str):
            if "minLength" in subschema and len(value) < subschema["minLength"]:
                errors.append(f"{key}: minLength 违规"
                              f"（长度 {len(value)} < {subschema['minLength']}）")
            if "maxLength" in subschema and len(value) > subschema["maxLength"]:
                errors.append(f"{key}: maxLength 违规"
                              f"（长度 {len(value)} > {subschema['maxLength']}）")
            # pattern
            if "pattern" in subschema and not re.search(subschema["pattern"], value):
                errors.append(f"{key}: pattern 违规（不匹配 {subschema['pattern']}）")

        # multipleOf（Decimal 精确判定）
        if "multipleOf" in subschema and isinstance(value, (int, float)):
            if not _check_multiple_of(value, subschema["multipleOf"]):
                errors.append(f"{key}: multipleOf 违规"
                              f"（{value} 不是 {subschema['multipleOf']} 的整数倍）")

        # items（数组元素逐个校验）
        if isinstance(value, list) and "items" in subschema:
            item_schema = subschema["items"]
            for i, item in enumerate(value):
                if "type" in item_schema and not _check_type(item, item_schema["type"]):
                    errors.append(f"{key}[{i}]: items type 违规"
                                  f"（期望 {item_schema['type']}，"
                                  f"实际 {type(item).__name__}）")

        # uniqueItems
        if subschema.get("uniqueItems") and isinstance(value, list):
            seen = []
            for item in value:
                marker = json.dumps(item, sort_keys=True, ensure_ascii=False)
                if marker in seen:
                    errors.append(f"{key}: uniqueItems 违规（存在重复元素）")
                    break
                seen.append(marker)

    return errors


if __name__ == "__main__":
    import doctest
    doctest.testmod(verbose=True)
```

## 设计要点

| Schema 约束 | 实现策略 |
|---|---|
| `multipleOf: 0.01` | `Decimal(str(value)) % Decimal("0.01")`——规避 `0.1 % 0.01` 浮点误差陷阱 |
| `type: ["array", "null"]` | 联合类型逐个匹配，`None` 通过 |
| `pattern: ^[A-Z]{3}$` | `re.search`（draft-07 语义为非锚定搜索，pattern 本身已含锚点） |
| `uniqueItems: true` | `json.dumps(sort_keys=True)` 序列化后判重，兼容混合类型数组 |
| 类型不符 | 跳过该字段后续断言，避免级联噪音 |
| `bool` 排除 | Python 中 `bool` 是 `int` 子类，`isinstance(True, number)` 需显式排除 |
