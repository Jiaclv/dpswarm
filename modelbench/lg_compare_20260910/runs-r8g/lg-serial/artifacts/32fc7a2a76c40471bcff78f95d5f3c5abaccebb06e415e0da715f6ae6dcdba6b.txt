    import json
    import re
    from fractions import Fraction

    SCHEMA_TEXT = """<A 的 json 全文，逐字，已核验一致>"""
    SCHEMA = json.loads(SCHEMA_TEXT)

    def validate_record(record: dict, schema: dict) -> list[str]:
        r"""
        >>> validate_record({"id": "a1", "amount": 12.34,
        ...                  "currency": "CNY", "tags": ["food", None.__class__ and "x"]}, SCHEMA)  # 见用例①实参
        []
        >>> validate_record({"id": "a1", "amount": 3.141,
        ...                  "currency": "CNY", "tags": None}, SCHEMA)
        ['amount: not a multiple of 0.01 (two decimal places required)']
        >>> validate_record({"id": "a1", "amount": 1.0,
        ...                  "currency": "cny", "tags": None}, SCHEMA)
        ['currency: does not match ^[A-Z]{3}$']
        """
        errors: list[str] = []
        props = schema.get("properties", {})
        for key in schema.get("required", []):
            if key not in record:
                errors.append(f"{key}: missing required field")
        if schema.get("additionalProperties") is False:
            for key in record:
                if key not in props:
                    errors.append(f"{key}: unexpected field (additionalProperties=false)")
        for key, sub in props.items():
            if key not in record:
                continue
            v = record[key]
            types = sub["type"] if isinstance(sub["type"], list) else [sub["type"]]
            if v is None and "null" in types:
                continue
            if "string" in types and not isinstance(v, str):
                errors.append(f"{key}: type mismatch, expected string")
                continue
            if "number" in types and (isinstance(v, bool) or not isinstance(v, (int, float))):
                errors.append(f"{key}: type mismatch, expected number")
                continue
            if isinstance(v, str):
                if "minLength" in sub and len(v) < sub["minLength"]:
                    errors.append(f"{key}: shorter than minLength {sub['minLength']}")
                if "pattern" in sub and not re.search(sub["pattern"], v):
                    errors.append(f"{key}: does not match {sub['pattern']}")
            if isinstance(v, (int, float)) and not isinstance(v, bool):
                if "multipleOf" in sub:
                    q = Fraction(str(v)) / Fraction(str(sub["multipleOf"]))
                    if q.denominator != 1:
                        errors.append(f"{key}: not a multiple of {sub['multipleOf']} (two decimal places required)")
            if isinstance(v, list):
                if not all(isinstance(x, str) for x in v):
                    errors.append(f"{key}: items must all be strings")
        return errors
