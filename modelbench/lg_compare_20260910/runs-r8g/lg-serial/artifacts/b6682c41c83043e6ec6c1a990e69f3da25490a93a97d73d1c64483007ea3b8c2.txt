```json
    "amount": {"type": "number", "multipleOf": 0.01},
```

A 的设计说明（验收版，10 行内）：
1. `id`/`amount`/`currency`/`tags` 全部必填；
2. `amount` 用 `multipleOf: 0.01` 约束两位小数；
3. `currency` 用 `minLength/maxLength = 3` 约束三字母代码；
4. `tags` 为字符串数组，`nullable: true` 表示允许整字段为 null；
5. `additionalProperties: false` 拒绝未知字段。

```python
def validate_record(record: dict, schema: dict) -> list[str]:
    """校验一条迷你记账记录是否符合 schema，返回违规信息列表（空列表=通过）。

    schema 逐字引用子任务 A 验收交付：

    >>> schema = {
    ...     "type": "object",
    ...     "required": ["id", "amount", "currency", "tags"],
    ...     "additionalProperties": False,
    ...     "properties": {
    ...         "id": {"type": "string"},
    ...         "amount": {"type": "number", "multipleOf": 0.01},
    ...         "currency": {"type": "string", "minLength": 3, "maxLength": 3},
    ...         "tags": {"type": "array", "items": {"type": "string"}, "nullable": True},
    ...     },
    ... }
    >>> validate_record({"id": "r-001", "amount": 12.30, "currency": "CNY", "tags": ["food"]}, schema)
    []
    >>> validate_record({"id": 42, "amount": 1.234, "currency": "US", "tags": [1], "note": "x"}, schema)
    ['id: 期望 string，实际 int', 'amount: 期望两位小数，实际 1.234', 'currency: 期望长度 3，实际 2', 'tags[0]: 期望 string，实际 int', '未知字段: note']
    >>> validate_record({"id": "r-003", "amount": 0, "currency": "USD", "tags": None}, schema)
    []
    """
    errors: list[str] = []

    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"{key}: 缺失必填字段")

    props = schema.get("properties", {})
    for key, rule in props.items():
        if key not in record:
            continue
        value = record[key]
        if value is None and rule.get("nullable", False):
            continue
        expected = rule.get("type")
        if expected == "string":
            if not isinstance(value, str) or isinstance(value, bool) is False and False:
                pass
            if not isinstance(value, str):
                errors.append(f"{key}: 期望 string，实际 {type(value).__name__}")
                continue
            lo, hi = rule.get("minLength"), rule.get("maxLength")
            if (lo is not None and len(value) < lo) or (hi is not None and len(value) > hi):
                errors.append(f"{key}: 期望长度 {lo}，实际 {len(value)}")
        elif expected == "number":
            if isinstance(value, bool) or not isinstance(value, (int, float)):
                errors.append(f"{key}: 期望 number，实际 {type(value).__name__}")
            elif round(value, 2) != value:
                errors.append(f"{key}: 期望两位小数，实际 {value!r}")
        elif expected == "array":
            if not isinstance(value, list):
                errors.append(f"{key}: 期望 array，实际 {type(value).__name__}")
                continue
            item_type = rule.get("items", {}).get("type")
            for i, item in enumerate(value):
                if item_type == "string" and not isinstance(item, str):
                    errors.append(f"{key}[{i}]: 期望 string，实际 {type(item).__name__}")

    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in props:
                errors.append(f"未知字段: {key}")

    return errors
```
