import re

SCHEMA_TEXT = """{"$schema":"https://json-schema.org/draft/2020-12/schema","title":"MiniBookkeepingRecord","type":"object","properties":{"id":{"type":"string","minLength":1},"amount":{"type":"number","multipleOf":0.01},"currency":{"type":"string","pattern":"^[A-Z]{3}$"},"tags":{"type":["array","null"],"items":{"type":"string"}}},"required":["id","amount","currency","tags"],"additionalProperties":false}"""
# ↑↑↑ 逐字内嵌结束 ↑↑↑


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表=通过）。

    >>> import json
    >>> schema = json.loads(SCHEMA_TEXT)
    >>> validate_record({"id": "rec-001", "amount": 12.30, "currency": "CNY", "tags": ["food", "travel"]}, schema)
    []
    >>> "amount" in "; ".join(validate_record({"id": "rec-002", "amount": 3.141, "currency": "CNY", "tags": []}, schema))
    True
    >>> "currency" in "; ".join(validate_record({"id": "rec-003", "amount": 9.99, "currency": "cny", "tags": None}, schema))
    True
    """
    errors: list[str] = []
    properties = schema.get("properties", {})

    # 1) required：缺失必填字段
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"missing required field: {key}")

    # 2) additionalProperties=false：拒绝未定义字段
    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in properties:
                errors.append(f"additional property not allowed: {key}")

    for key, value in record.items():
        if key not in properties:
            continue
        sub = properties[key]
        declared = sub.get("type")
        allowed = declared if isinstance(declared, list) else [declared]

        # 3) 类型检查（bool 不得冒充 number/string/array）
        if isinstance(value, bool):
            type_ok = False
        elif "null" in allowed and value is None:
            type_ok = True
        elif "string" in allowed and isinstance(value, str):
            type_ok = True
        elif "number" in allowed and isinstance(value, (int, float)):
            type_ok = True
        elif "array" in allowed and isinstance(value, list):
            type_ok = True
        else:
            type_ok = False
        if not type_ok:
            errors.append(f"type mismatch for {key}: expected {allowed}, got {type(value).__name__}")
            continue

        # 4) id：minLength 1（空串违规）
        if key == "id" and isinstance(value, str) and len(value) < sub.get("minLength", 0):
            errors.append("id must not be an empty string")

        # 5) amount：数值且两位小数（multipleOf 0.01，Decimal 避免浮点误差）
        if key == "amount" and "multipleOf" in sub:
            step = Decimal(str(sub["multipleOf"]))
            if Decimal(str(value)) % step != 0:
                errors.append(
                    f"amount must be a number with at most 2 decimal places (multipleOf {sub['multipleOf']}), got {value}"
                )

        # 6) currency：^[A-Z]{3}$
        if key == "currency" and "pattern" in sub:
            if not re.fullmatch(sub["pattern"], value):
                errors.append(
                    f"currency must match pattern {sub['pattern']} (3 uppercase letters), got {value!r}"
                )

        # 7) tags：字符串数组或 null；元素必须全为字符串
        if key == "tags" and isinstance(value, list):
            for i, item in enumerate(value):
                if not isinstance(item, str) or isinstance(item, bool):
                    errors.append(f"tags[{i}] must be a string, got {type(item).__name__}")

    return errors
