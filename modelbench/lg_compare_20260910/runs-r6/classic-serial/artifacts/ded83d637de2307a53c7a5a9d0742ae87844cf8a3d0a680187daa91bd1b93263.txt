## 完整交付物（纯标准库）

```python
import re
from decimal import Decimal

_TYPE_CHECKS = {
    "string": lambda v: isinstance(v, str),
    "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
    "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
    "boolean": lambda v: isinstance(v, bool),
    "array": lambda v: isinstance(v, list),
    "object": lambda v: isinstance(v, dict),
    "null": lambda v: v is None,
}


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验记账记录是否符合 schema，返回违规信息列表；空列表 = 通过。

    schema 判定依据与 wi-cc8152c306 已验收交付的 json schema 块逐字一致
    （id: string/minLength；amount: number/multipleOf 0.01；
    currency: string/pattern ^[A-Z]{3}$；tags: [array, null]/items string，可空）。

    >>> schema = {
    ...     "type": "object",
    ...     "properties": {
    ...         "id": {"type": "string", "minLength": 1},
    ...         "amount": {"type": "number", "multipleOf": 0.01},
    ...         "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
    ...         "tags": {"type": ["array", "null"], "items": {"type": "string"}}
    ...     },
    ...     "required": ["id", "amount", "currency", "tags"]
    ... }
    >>> validate_record({"id": "r-001", "amount": 12.5, "currency": "CNY", "tags": ["food", "lunch"]}, schema)
    []
    >>> validate_record({"id": "r-002", "amount": 3.141, "currency": "USD", "tags": None}, schema)
    ['amount: 值 3.141 不是 0.01 的整数倍']
    >>> validate_record({"id": "r-003", "amount": 9.99, "currency": "usd", "tags": ["food", 7]}, schema)
    ['currency: 值 usd 不匹配模式 ^[A-Z]{3}$', 'tags: 第 2 个元素(7)不是 string']
    """
    problems: list[str] = []

    for field in schema.get("required", []):
        if field not in record:
            problems.append(f"{field}: 缺少必填字段")

    for field, spec in schema.get("properties", {}).items():
        if field not in record:
            continue
        value = record[field]

        # 1) type 判定（字符串形式或 [array, null] 列表形式）
        raw_type = spec.get("type")
        type_names = [raw_type] if isinstance(raw_type, str) else list(raw_type or [])
        checks = [_TYPE_CHECKS[t] for t in type_names if t in _TYPE_CHECKS]
        if checks and not any(check(value) for check in checks):
            problems.append(f"{field}: 值 {value} 不是 {type_names} 类型")
            continue

        # 2) minLength：仅对字符串生效
        if isinstance(value, str) and "minLength" in spec and len(value) < spec["minLength"]:
            problems.append(f"{field}: 长度 {len(value)} 小于 minLength {spec['minLength']}")

        # 3) multipleOf：Decimal(str(x)) 判定，规避浮点误差（3.141 % 0.01 = 0.001 → 违规）
        if "multipleOf" in spec and isinstance(value, (int, float)) and not isinstance(value, bool):
            if Decimal(str(value)) % Decimal(str(spec["multipleOf"])) != 0:
                problems.append(f"{field}: 值 {value} 不是 {spec['multipleOf']} 的整数倍")

        # 4) pattern：按 JSON Schema 语义做搜索式正则匹配
        if isinstance(value, str) and "pattern" in spec and re.search(spec["pattern"], value) is None:
            problems.append(f"{field}: 值 {value} 不匹配模式 {spec['pattern']}")

        # 5) items：数组元素逐个判定类型（tags 为 null 时因非 list 自动跳过）
        if isinstance(value, list) and "items" in spec:
            item_type = spec["items"].get("type")
            item_check = _TYPE_CHECKS.get(item_type)
            if item_check:
                for i, item in enumerate(value):
                    if not item_check(item):
                        problems.append(f"{field}: 第 {i + 1} 个元素({item})不是 {item_type}")

    return problems
```
