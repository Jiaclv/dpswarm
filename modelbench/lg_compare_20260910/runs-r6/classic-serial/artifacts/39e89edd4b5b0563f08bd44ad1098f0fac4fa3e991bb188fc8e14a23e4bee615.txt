## 实现（完整，纯标准库）

```python
import re
from decimal import Decimal

# SCHEMA 常量 = 上方 ```json 代码块逐字内容（dict 形式，字段/约束一一对应，无增删改）
SCHEMA = {
    "$schema": "https://json-schema.org/draft/2020-12/schema",
    "title": "迷你记账记录",
    "type": "object",
    "properties": {
        "id": {"type": "string", "minLength": 1},
        "amount": {"type": "number", "multipleOf": 0.01},
        "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
        "tags": {"type": ["array", "null"], "items": {"type": "string"}, "uniqueItems": True},
    },
    "required": ["id", "amount", "currency", "tags"],
}


def validate_record(record: dict, schema: dict) -> list[str]:
    """按 schema 校验 record，返回违规信息列表（空列表=通过）。

    >>> validate_record({"id": "r001", "amount": 12.5, "currency": "CNY", "tags": ["餐饮", "日常"]}, SCHEMA)
    []

    >>> validate_record({"id": "", "amount": 12.345, "currency": "cn1", "tags": ["a", "a"]}, SCHEMA)
    ['id: 长度 0 低于 minLength=1', 'amount: 12.345 不是 0.01 的整数倍（不满足两位小数）', 'currency: 'cn1' 不匹配 ^[A-Z]{3}$', 'tags: 存在重复元素']

    >>> validate_record({"id": "r002", "amount": 0.01, "currency": "USD", "tags": None}, SCHEMA)
    []
    """
    errors: list[str] = []
    props = schema.get("properties", {})

    for field in schema.get("required", []):
        if field not in record:
            errors.append(f"{field}: 缺失必填字段")

    for field, rule in props.items():
        if field not in record:
            continue
        value = record[field]

        declared = rule.get("type")
        types = [declared] if isinstance(declared, str) else list(declared or [])
        is_bool = isinstance(value, bool)
        if isinstance(value, str) and "string" in types:
            pass
        elif is_bool and "number" in types:
            errors.append(f"{field}: bool 不视为 number")
            continue
        elif isinstance(value, (int, float)) and not is_bool and "number" in types:
            pass
        elif isinstance(value, list) and "array" in types:
            pass
        elif value is None and "null" in types:
            continue
        else:
            errors.append(f"{field}: 类型 {type(value).__name__} 不在允许范围 {types}")
            continue

        if isinstance(value, str):
            min_len = rule.get("minLength")
            if min_len is not None and len(value) < min_len:
                errors.append(f"{field}: 长度 {len(value)} 低于 minLength={min_len}")
            pattern = rule.get("pattern")
            if pattern is not None and not re.search(pattern, value):
                errors.append(f"{field}: {value!r} 不匹配 {pattern}")

        elif isinstance(value, (int, float)) and not isinstance(value, bool):
            multiple = rule.get("multipleOf")
            if multiple is not None:
                if Decimal(str(value)) % Decimal(str(multiple)) != 0:
                    errors.append(f"{field}: {value} 不是 {multiple} 的整数倍（不满足两位小数）")

        elif isinstance(value, list):
            if rule.get("uniqueItems") and len({repr(x) for x in value}) != len(value):
                errors.append(f"{field}: 存在重复元素")
            item_rule = rule.get("items", {})
            item_types = item_rule.get("type")
            if item_types == "string":
                for i, el in enumerate(value):
                    if not isinstance(el, str):
                        errors.append(f"{field}[{i}]: 元素类型 {type(el).__name__} 非 string")

    return errors
