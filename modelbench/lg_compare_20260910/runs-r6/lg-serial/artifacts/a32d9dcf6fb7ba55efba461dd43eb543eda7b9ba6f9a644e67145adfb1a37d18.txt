import json
import re
from fractions import Fraction

# 逐字引用子任务 A（[wi-3ef26a8a60]）交付的 schema 全文：
SCHEMA_TEXT = '''{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "title": "MiniBookkeepingRecord",
  "type": "object",
  "properties": {
    "id": {
      "type": "string",
      "minLength": 1
    },
    "amount": {
      "type": "number",
      "multipleOf": 0.01
    },
    "currency": {
      "type": "string",
      "pattern": "^[A-Z]{3}$"
    },
    "tags": {
      "type": ["array", "null"],
      "items": { "type": "string", "minLength": 1 },
      "uniqueItems": true
    }
  },
  "required": ["id", "amount", "currency", "tags"],
  "additionalProperties": false
}'''


def validate_record(record: dict, schema: dict) -> list[str]:
    """按 schema 校验单条记录，返回违规信息列表（空列表=通过）。

    纯标准库实现，覆盖本 schema 用到的关键字：type（含类型数组）、
    properties、required、additionalProperties、minLength、pattern、
    multipleOf、items、uniqueItems。

    >>> SCHEMA = json.loads(SCHEMA_TEXT)
    >>> validate_record({"id": "r001", "amount": 12.34,
    ...                  "currency": "CNY", "tags": ["food", "lunch"]}, SCHEMA)
    []
    >>> validate_record({"id": "", "amount": 12.345,
    ...                  "currency": "cny", "tags": ["food", "food"]}, SCHEMA)
    ['id: minLength 1 violated (got length 0)', 'amount: not a multiple of 0.01', "currency: does not match pattern '^[A-Z]{3}$'", 'tags: duplicate items at positions 0, 1']
    >>> validate_record({"id": "r002", "amount": 5,
    ...                  "currency": "USD", "extra": 1}, SCHEMA)
    ['tags: required field missing', 'extra: additional property not allowed']
    """
    errors: list[str] = []

    def type_ok(value, t):
        if t == "object":
            return isinstance(value, dict)
        if t == "array":
            return isinstance(value, list)
        if t == "string":
            return isinstance(value, str)
        if t == "number":
            return isinstance(value, (int, float)) and not isinstance(value, bool)
        if t == "null":
            return value is None
        return False

    # 顶层 object 校验
    top_type = schema.get("type", "object")
    allowed = top_type if isinstance(top_type, list) else [top_type]
    if not any(type_ok(record, t) for t in allowed):
        return [f"record: expected type {allowed}, got {type(record).__name__}"]

    props = schema.get("properties", {})
    for name in schema.get("required", []):
        if name not in record:
            errors.append(f"{name}: required field missing")
    if schema.get("additionalProperties") is False:
        for name in record:
            if name not in props:
                errors.append(f"{name}: additional property not allowed")

    for name, sub in props.items():
        if name not in record:
            continue
        value = record[name]
        stype = sub.get("type")
        sallowed = stype if isinstance(stype, list) else [stype]
        if stype is not None and not any(type_ok(value, t) for t in sallowed):
            errors.append(
                f"{name}: expected type {sallowed}, got {type(value).__name__}")
            continue
        if isinstance(value, str):
            if "minLength" in sub and len(value) < sub["minLength"]:
                errors.append(
                    f"{name}: minLength {sub['minLength']} violated (got length {len(value)})")
            if "pattern" in sub and not re.search(sub["pattern"], value):
                errors.append(f"{name}: does not match pattern '{sub['pattern']}'")
        elif isinstance(value, (int, float)) and not isinstance(value, bool):
            if "multipleOf" in sub:
                # 用 Fraction 规避二进制浮点精度问题（A 设计说明第 1 条）
                if Fraction(str(value)) % Fraction(str(sub["multipleOf"])) != 0:
                    errors.append(f"{name}: not a multiple of {sub['multipleOf']}")
        elif isinstance(value, list):
            if sub.get("uniqueItems"):
                seen = {}
                for i, item in enumerate(value):
                    key = json.dumps(item, sort_keys=True, ensure_ascii=False)
                    if key in seen:
                        errors.append(
                            f"{name}: duplicate items at positions {seen[key]}, {i}")
                    else:
                        seen[key] = i
            items_schema = sub.get("items")
            if isinstance(items_schema, dict):
                for i, item in enumerate(value):
                    if isinstance(item, str):
                        if "minLength" in items_schema and len(item) < items_schema["minLength"]:
                            errors.append(
                                f"{name}[{i}]: minLength {items_schema['minLength']} violated (got length {len(item)})")
                        if "pattern" in items_schema and not re.search(items_schema["pattern"], item):
                            errors.append(
                                f"{name}[{i}]: does not match pattern '{items_schema['pattern']}'")
    return errors
