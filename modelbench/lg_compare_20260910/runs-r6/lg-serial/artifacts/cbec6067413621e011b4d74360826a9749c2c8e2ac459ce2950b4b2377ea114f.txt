SCHEMA_TEXT = """{

  "required": ["id", "amount", "currency", "tags"]

def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表 = 通过）。

    >>> validate_record({"id": "a1", "amount": 12.34, "currency": "CNY",
    ...                  "tags": ["food", "travel"]}, json.loads(SCHEMA_TEXT))
    []
    >>> validate_record({"id": "a1", "amount": 12.34,
    ...                  "currency": "CNY"}, json.loads(SCHEMA_TEXT))
    ['tags: required field missing']
    >>> validate_record({"id": "", "amount": "12.34", "currency": "CNY",
    ...                  "tags": ["food", 7]}, json.loads(SCHEMA_TEXT))
    ['id: minLength 1 violated', 'amount: expected number, got str', 'tags[1]: expected string, got int']
    """
    errors: list[str] = []
    # 先查 required（按 schema 声明顺序，保证输出确定性）
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"{key}: required field missing")
    # 再逐字段按 properties 声明顺序校验
    for key, sub in schema.get("properties", {}).items():
        if key in record:
            errors.extend(_validate_value(key, record[key], sub))
    return errors


def _validate_value(path: str, value, sub: dict) -> list[str]:
    errors: list[str] = []
    expected = sub.get("type")
    if expected is not None:
        allowed = expected if isinstance(expected, list) else [expected]
        if not any(_type_ok(value, t) for t in allowed):
            errors.append(
                f"{path}: expected {' or '.join(allowed)}, got {type(value).__name__}"
            )
            return errors  # 类型不符时不再深入检查
    if "minLength" in sub and isinstance(value, str):
        if len(value) < sub["minLength"]:
            errors.append(f"{path}: minLength {sub['minLength']} violated")
    if "items" in sub and isinstance(value, list):
        for i, item in enumerate(value):
            errors.extend(_validate_value(f"{path}[{i}]", item, sub["items"]))
    if "properties" in sub and isinstance(value, dict):
        for k, s in sub["properties"].items():
            if k in value:
                errors.extend(_validate_value(f"{path}.{k}", value[k], s))
    return errors


def _type_ok(value, t: str) -> bool:
    checks = {
        "string": lambda v: isinstance(v, str),
        "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
        "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
        "boolean": lambda v: isinstance(v, bool),
        "array": lambda v: isinstance(v, list),
        "object": lambda v: isinstance(v, dict),
        "null": lambda v: v is None,
    }
    return checks[t](value)
