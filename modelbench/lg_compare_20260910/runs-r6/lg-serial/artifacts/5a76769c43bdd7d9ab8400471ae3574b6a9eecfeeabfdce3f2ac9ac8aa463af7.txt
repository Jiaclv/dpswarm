"""迷你记账记录校验器（子任务 B）。

严格依据子任务 A 验收后的 schema 文本（SCHEMA_TEXT，逐字引用）实现
validate_record(record: dict, schema: dict) -> list[str]。
仅用 Python 标准库，无第三方依赖。
"""

import re

# 子任务 A 交付的 schema 全文（逐字引用，勿改动）
SCHEMA_TEXT = """{
  "type": "object",
  "properties": {
    "id": {"type": "string"},
    "amount": {"type": "number", "multipleOf": 0.01},
    "currency": {"type": "string", "minLength": 3, "pattern": "^[A-Z]{3}$"},
    "tags": {
      "type": "array",
      "items": {"type": "string"},
      "uniqueItems": true
    }
  },
  "required": ["id", "amount", "currency"],
  "additionalProperties": false
}"""


def type_ok(value, expected: str) -> bool:
    """按 JSON Schema 基本类型判断 value 是否匹配。"""
    if expected == "object":
        return isinstance(value, dict)
    if expected == "array":
        return isinstance(value, list)
    if expected == "string":
        return isinstance(value, str)
    if expected == "number":
        return isinstance(value, (int, float)) and not isinstance(value, bool)
    if expected == "integer":
        return isinstance(value, int) and not isinstance(value, bool)
    if expected == "boolean":
        return isinstance(value, bool)
    if expected == "null":
        return value is None
    return False


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表=通过）。

    >>> import json
    >>> schema = json.loads(SCHEMA_TEXT)
    >>> validate_record({"id": "r1", "amount": 12.5,
    ...                  "currency": "CNY", "tags": []}, schema)
    []
    >>> validate_record({"id": "r2", "amount": 1.234,
    ...                  "currency": "usa", "tags": ["a", "a"],
    ...                  "note": "extra"}, schema)
    ['currency: minLength 3 violated', 'currency: pattern ^[A-Z]{3}$ violated', 'amount: multipleOf 0.01 violated', 'tags: uniqueItems violated', 'additionalProperties: unexpected key note']
    >>> validate_record({"amount": 3.0, "currency": "USD"}, schema)
    ['required: missing id']
    """
    errors: list[str] = []

    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"required: missing {key}")

    properties = schema.get("properties", {})
    for key, sub in properties.items():
        if key not in record:
            continue
        value = record[key]
        if "type" in sub and not type_ok(value, sub["type"]):
            errors.append(f"{key}: type {sub['type']} violated")
            continue
        if "minLength" in sub and len(value) < sub["minLength"]:
            errors.append(f"{key}: minLength {sub['minLength']} violated")
        if "pattern" in sub and not re.search(sub["pattern"], value):
            errors.append(f"{key}: pattern {sub['pattern']} violated")
        if "multipleOf" in sub:
            m = sub["multipleOf"]
            if m == 0 or round(value / m) != value / m:
                errors.append(f"{key}: multipleOf {m} violated")
        if "items" in sub and isinstance(value, list):
            for i, item in enumerate(value):
                if not type_ok(item, sub["items"].get("type", "")):
                    errors.append(f"{key}: items[{i}] type violated")
        if sub.get("uniqueItems") and isinstance(value, list):
            seen = []
            for item in value:
                if item in seen:
                    errors.append(f"{key}: uniqueItems violated")
                    break
                seen.append(item)

    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in properties:
                errors.append(f"additionalProperties: unexpected key {key}")

    return errors

    doctest.testmod(verbose=False)
