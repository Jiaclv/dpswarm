### 二、交付正文（纯标准库，可直接实跑）

```python
"""validate_record —— 迷你记账记录校验器（纯标准库）。

_A_SCHEMA 为上游 wi-bf3842080d（子任务 A）验收 schema 全文的逐字引用。
"""

import json
import re
from fractions import Fraction

_A_SCHEMA = json.loads(r'''
{
  "type": "object",
  "properties": {
    "id": {"type": "string", "minLength": 1},
    "amount": {"type": "number", "multipleOf": 0.01},
    "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
    "tags": {"type": "array", "items": {"type": "string"}}
  },
  "required": ["id", "amount", "currency", "tags"],
  "additionalProperties": false
}
''')


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表 = 通过）。

    >>> validate_record({"id": "r1", "amount": 10.5, "currency": "USD", "tags": []}, _A_SCHEMA)
    []
    >>> validate_record({"id": "r2", "amount": 7.005, "currency": "CNY", "tags": []}, _A_SCHEMA)
    ['amount: 不是 0.01 的整数倍']
    >>> validate_record({"id": "r3", "amount": 1.25, "currency": "us12", "tags": [1, "food"]}, _A_SCHEMA)
    ['currency: 不匹配 pattern', 'tags[0]: 应为 string']
    """
    errors: list[str] = []
    if not isinstance(record, dict):
        return ["record: 应为 object"]
    props = schema.get("properties", {})
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"{key}: 缺少必填字段")
    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in props:
                errors.append(f"{key}: 不允许的额外字段")
    for key, spec in props.items():
        if key in record:
            _validate_value(key, record[key], spec, errors)
    return errors


def _validate_value(key: str, value, spec: dict, errors: list[str]) -> None:
    t = spec.get("type")
    if t == "string":
        if not isinstance(value, str):
            errors.append(f"{key}: 应为 string")
            return
        if "minLength" in spec and len(value) < spec["minLength"]:
            errors.append(f"{key}: 长度小于 minLength={spec['minLength']}")
        if "pattern" in spec and not re.search(spec["pattern"], value):
            errors.append(f"{key}: 不匹配 pattern")
    elif t == "number":
        if isinstance(value, bool) or not isinstance(value, (int, float)):
            errors.append(f"{key}: 应为 number")
            return
        if "multipleOf" in spec:
            m = Fraction(str(spec["multipleOf"]))
            # 字符串精确转换做整除判定，规避浮点误差（A 设计说明第 1 条）
            if Fraction(str(value)) % m != 0:
                errors.append(f"{key}: 不是 {spec['multipleOf']} 的整数倍")
    elif t == "array":
        if not isinstance(value, list):
            errors.append(f"{key}: 应为 array")
            return
        items = spec.get("items")
        if items:
            for i, item in enumerate(value):
                _validate_value(f"{key}[{i}]", item, items, errors)


if __name__ == "__main__":
    import doctest
    result = doctest.testmod()
    print(f"doctest: {result.attempted} 项，失败 {result.failed} 项")
```
