### validate_record 实现（针对上一轮全部技术缺陷已修正）

```python
import re
from decimal import Decimal


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表=通过）。
    支持本轮已确认的 A 约束：type 联合类型（含 null）、minLength、
    required、additionalProperties:false、pattern、multipleOf(Decimal)。
    """
    errors: list[str] = []
    props = schema.get("properties", {})
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"缺少必填字段: {key}")
    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in props:
                errors.append(f"多余字段: {key}")
    for key, sub in props.items():
        if key in record:
            errors.extend(_check(key, record[key], sub))
    return errors


def _check(key, value, sub):
    errs: list[str] = []
    declared = sub.get("type")
    allowed = declared if isinstance(declared, list) else [declared]  # 联合类型
    if not any(_matches(value, t) for t in allowed):
        return [f"{key}: 类型须为 {allowed}"]
    if value is None:
        return errs
    if isinstance(value, str):
        if len(value) < sub.get("minLength", 0):
            errs.append(f"{key}: 长度须≥{sub['minLength']}")
        if "pattern" in sub and not re.search(sub["pattern"], value):
            errs.append(f"{key}: 不匹配 {sub['pattern']}")
    if isinstance(value, list):
        item_schema = sub.get("items", {})
        for i, item in enumerate(value):
            errs.extend(f"{key}[{i}]: {m}" for m in _check(f"{key}[{i}]", item, item_schema))
    if "multipleOf" in sub and isinstance(value, (int, float)) and not isinstance(value, bool):
        # 依 A 设计说明：Decimal 校验，不用浮点容差
        if Decimal(str(value)) % Decimal(str(sub["multipleOf"])) != 0:
            errs.append(f"{key}: 须为 {sub['multipleOf']} 的整数倍")
    return errs


def _matches(value, t):
    return {"string": lambda v: isinstance(v, str),
            "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
            "array": lambda v: isinstance(v, list),
            "null": lambda v: v is None,
            "object": lambda v: isinstance(v, dict)}.get(t, lambda v: False)(value)
```
