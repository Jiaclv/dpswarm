import re


def validate_record(record: dict, schema: dict) -> list[str]:
    """依据 schema 校验 record，返回违规信息列表（空列表 = 通过）。

    >>> schema = {
    ...     "$schema": "http://json-schema.org/draft-07/schema#",
    ...     "title": "MiniLedgerRecord",
    ...     "type": "object",
    ...     "properties": {
    ...         "id": {"type": "string", "minLength": 1},
    ...         "amount": {"type": "number", "minimum": 0, "multipleOf": 0.01},
    ...         "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
    ...         "tags": {"type": ["array", "null"], "items": {"type": "string"}}
    ...     },
    ...     "required": ["id", "amount", "currency"],
    ...     "additionalProperties": False
    ... }
    >>> validate_record({"id": "r1", "amount": 12.50, "currency": "USD", "tags": None}, schema)
    []
    >>> validate_record({"id": "", "amount": -1, "currency": "usd", "tags": [1]}, schema)
    ['id: string length 0 is less than minLength 1', 'amount: -1 is less than minimum 0', 'currency: does not match pattern ^[A-Z]{3}$', 'tags[0]: item is not a string']
    >>> validate_record({"amount": 5, "note": "hi"}, schema)
    ['missing required field: id', 'missing required field: currency', 'unexpected field: note']
    """
    errors: list[str] = []

    # 1) 必填字段
    for prop in schema.get("required", []):
        if prop not in record:
            errors.append(f"missing required field: {prop}")

    properties = schema.get("properties", {})
    for key, value in record.items():
        spec = properties.get(key)
        if spec is None:
            if schema.get("additionalProperties") is False:
                errors.append(f"unexpected field: {key}")
            continue

        declared = spec.get("type")
        type_list = declared if isinstance(declared, list) else [declared]

        if value is None:
            if "null" not in type_list:
                errors.append(f"{key}: null is not allowed")
            continue

        if "string" in type_list:
            if not isinstance(value, str):
                errors.append(f"{key}: expected string, got {type(value).__name__}")
                continue
            if "minLength" in spec and len(value) < spec["minLength"]:
                errors.append(
                    f"{key}: string length {len(value)} is less than minLength {spec['minLength']}"
                )
            if "pattern" in spec and re.search(spec["pattern"], value) is None:
                errors.append(f"{key}: does not match pattern {spec['pattern']}")

        elif "number" in type_list:
            if isinstance(value, bool) or not isinstance(value, (int, float)):
                errors.append(f"{key}: expected number, got {type(value).__name__}")
                continue
            if "minimum" in spec and value < spec["minimum"]:
                errors.append(f"{key}: {value} is less than minimum {spec['minimum']}")
            if "multipleOf" in spec:
                quotient = value / spec["multipleOf"]
                if abs(quotient - round(quotient)) > 1e-9:
                    errors.append(f"{key}: {value} is not a multiple of {spec['multipleOf']}")

        elif "array" in type_list:
            if not isinstance(value, list):
                errors.append(f"{key}: expected array, got {type(value).__name__}")
                continue
            item_spec = spec.get("items", {})
            for i, item in enumerate(value):
                if item_spec.get("type") == "string" and not isinstance(item, str):
                    errors.append(f"{key}[{i}]: item is not a string")

    return errors
