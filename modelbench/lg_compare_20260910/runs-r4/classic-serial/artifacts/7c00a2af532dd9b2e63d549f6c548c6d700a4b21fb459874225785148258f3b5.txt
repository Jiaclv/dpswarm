## 预交付 1：校验器实现（与 A 的 schema 文本无耦合，可先行验收）

纯标准库（`re` + `decimal`），schema 驱动、不硬编码字段名，A 最终产出何种字段约束均可覆盖：

```python
"""mini-ledger 校验器：validate_record(record, schema) -> list[str]"""
import re
from decimal import Decimal, InvalidOperation

_TYPE_MAP = {
    "string": str, "number": (int, float), "integer": int,
    "boolean": bool, "array": list, "object": dict,
}

def validate_record(record: dict, schema: dict) -> list[str]:
    """按 JSON Schema 子集校验 record，返回违规信息列表（空列表=通过）。

    支持：type / required / properties / pattern / minLength /
    maxLength / multipleOf（用于两位小数约束）/ items。
    """
    errors: list[str] = []

    def check(value, sch, path):
        t = sch.get("type")
        if t is not None:
            # bool 是 int 子类，须显式排除
            if t == "boolean":
                ok = isinstance(value, bool)
            else:
                ok = isinstance(value, _TYPE_MAP[t]) and not isinstance(value, bool)
            if not ok:
                errors.append(f"{path}: 期望类型 {t}, 实际 {type(value).__name__}")
                return
        if t in ("number", "integer") and "multipleOf" in sch:
            try:
                if Decimal(str(value)) % Decimal(str(sch["multipleOf"])) != 0:
                    errors.append(f"{path}: 不满足 multipleOf {sch['multipleOf']}")
            except InvalidOperation:
                errors.append(f"{path}: 无法进行数值取模检查")
        if isinstance(value, str):
            if "pattern" in sch and re.search(sch["pattern"], value) is None:
                errors.append(f"{path}: 不匹配 pattern {sch['pattern']!r}")
            if "minLength" in sch and len(value) < sch["minLength"]:
                errors.append(f"{path}: 长度 {len(value)} < minLength {sch['minLength']}")
            if "maxLength" in sch and len(value) > sch["maxLength"]:
                errors.append(f"{path}: 长度 {len(value)} > maxLength {sch['maxLength']}")
        if isinstance(value, list) and "items" in sch:
            for i, item in enumerate(value):
                check(item, sch["items"], f"{path}[{i}]")
        if isinstance(value, dict):
            for req in sch.get("required", []):
                if req not in value:
                    errors.append(f"{path}: 缺少必填字段 {req!r}")
            for key, sub in sch.get("properties", {}).items():
                if key in value:
                    check(value[key], sub, f"{path}.{key}")

    check(record, schema, "$")
    return errors
```
