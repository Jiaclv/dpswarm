"""子任务 B「校验器实现」——纯标准库，Python 3.8+"""

import re

# MINI_LEDGER_SCHEMA：A 交付文本逐字副本（仅 json.loads 语义，无任何改写；
# 保留顶层 $schema/$id/title、id.minLength=1、tags 可空 ["array","null"]）
MINI_LEDGER_SCHEMA = {
    "$schema": "https://json-schema.org/draft/2020-12/schema",
    "$id": "https://example.org/mini-ledger-record.schema.json",
    "title": "Mini Ledger Record",
    "type": "object",
    "properties": {
        "id": {"type": "string", "minLength": 1},
        "amount": {"type": "number", "multipleOf": 0.01},
        "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
        "tags": {"type": ["array", "null"], "items": {"type": "string"}}
    },
    "required": ["id", "amount", "currency", "tags"],
    "additionalProperties": False
}

_TYPE_MAP = {
    "string": (str,), "number": (int, float), "integer": (int,),
    "boolean": (bool,), "array": (list,), "object": (dict,), "null": (type(None),),
}


def validate_record(record: dict, schema: dict) -> list:
    """按 JSON Schema 校验单条记账记录，返回违规信息列表（空列表=通过）。

    支持本 schema 用到的关键字：required、additionalProperties、type（含可空列表）、
    minLength、pattern、multipleOf、items.type。违规按 schema 声明顺序产出。

    >>> validate_record({"id": "T1", "amount": 12.50, "currency": "CNY", "tags": ["food"]}, MINI_LEDGER_SCHEMA)
    []
    >>> validate_record({"id": "T2", "amount": -0.07, "currency": "USD", "tags": None}, MINI_LEDGER_SCHEMA)
    []
    >>> validate_record({"id": "", "amount": 1.234, "currency": "cn", "tags": ["food", 3], "note": "x"}, MINI_LEDGER_SCHEMA)
    ['id: length < minLength 1', 'amount: not a multiple of 0.01', 'currency: does not match pattern ^[A-Z]{3}$', 'tags[1]: expected type string', 'unexpected field: note']
    """
    errors = []
    props = schema.get("properties", {})

    # required
    for key in schema.get("required", []):
        if key not in record:
            errors.append("missing required field: %s" % key)

    additional = schema.get("additionalProperties", True)
    for key, value in record.items():
        if key not in props:
            if additional is False:
                errors.append("unexpected field: %s" % key)
            continue
        rule = props[key]
        types = rule.get("type")
        if types is not None:
            if isinstance(types, str):
                types = [types]
            py_types = tuple(t for name in types for t in _TYPE_MAP[name])
            # bool 是 int 子类，number 位排除 bool
            if not (isinstance(value, py_types)
                    and not (isinstance(value, bool) and "boolean" not in types)):
                errors.append("%s: expected type %s" % (key, "/".join(types)))
                continue
        if isinstance(value, str):
            if "minLength" in rule and len(value) < rule["minLength"]:
                errors.append("%s: length < minLength %s" % (key, rule["minLength"]))
            if "pattern" in rule and not re.search(rule["pattern"], value):
                errors.append("%s: does not match pattern %s" % (key, rule["pattern"]))
        if isinstance(value, (int, float)) and not isinstance(value, bool) and "multipleOf" in rule:
            q = value / rule["multipleOf"]
            if abs(q - round(q)) > 1e-9:
                errors.append("%s: not a multiple of %s" % (key, rule["multipleOf"]))
        if isinstance(value, list) and "items" in rule:
            item_type = rule["items"].get("type")
            if item_type:
                for i, item in enumerate(value):
                    if not isinstance(item, _TYPE_MAP[item_type]):
                        errors.append("%s[%d]: expected type %s" % (key, i, item_type))
    return errors
