## validate_record 实现（纯标准库）

```python
# validator.py
"""迷你记账记录校验器：严格按 JSON Schema 子集校验，返回违规信息列表（空列表=通过）。"""
import re
from decimal import Decimal

_TYPE_CHECK = {
    "string": lambda v: isinstance(v, str),
    "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
    "array": lambda v: isinstance(v, list),
    "object": lambda v: isinstance(v, dict),
    "boolean": lambda v: isinstance(v, bool),
    "null": lambda v: v is None,
}


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表；空列表表示通过。

    >>> validate_record(  # 用例 1：合法记录
    ...     {"id": "r-001", "amount": 12.50, "currency": "CNY", "tags": ["food", "lunch"]},
    ...     _SCHEMA)
    []

    >>> validate_record(  # 用例 2：amount 超两位小数 + currency 非 3 字母
    ...     {"id": "r-002", "amount": 1.234, "currency": "cny1", "tags": []},
    ...     _SCHEMA)
    ['amount: 数值须最多两位小数', 'currency: 不匹配模式 ^[A-Z]{3}$']

    >>> validate_record(  # 用例 3：缺 tags 字段 + tags 元素非字符串
    ...     {"id": "r-003", "amount": 9.99, "currency": "USD", "tags": [7]},
    ...     {"type": "object", "required": ["id", "amount", "currency", "tags"],
    ...      "properties": {"tags": {"type": "array", "items": {"type": "string"}}}})
    ['tags: 缺少必填字段', 'tags[0]: 期望 string，实际 int']

    >>> _ = _SCHEMA  # noqa: 占位引用，schema 全文见下方 ⚠ 块
    """
    errors: list[str] = []
    _validate(record, schema, path="", errors=errors)
    return errors


def _validate(value, schema: dict, path: str, errors: list[str]) -> None:
    label = path or "<root>"
    expected = schema.get("type")
    if expected and not _TYPE_CHECK.get(expected, lambda v: True)(value):
        errors.append(f"{label}: 期望 {expected}，实际 {type(value).__name__}")
        return

    if expected == "object":
        for key in schema.get("required", []):
            if key not in value:
                errors.append(f"{label}{key}: 缺少必填字段".replace("<root>", "") if not path
                              else f"{label}.{key}: 缺少必填字段")
        for key, sub in schema.get("properties", {}).items():
            if key in value:
                _validate(value[key], sub, f"{path}.{key}" if path else key, errors)

    elif expected == "array":
        for i, item in enumerate(value):
            if "items" in schema:
                _validate(item, schema["items"], f"{path}[{i}]", errors)

    elif expected == "string":
        if "pattern" in schema and re.search(schema["pattern"], value) is None:
            errors.append(f"{label}: 不匹配模式 {schema['pattern']}")
        if "minLength" in schema and len(value) < schema["minLength"]:
            errors.append(f"{label}: 长度须 >= {schema['minLength']}")
        if "maxLength" in schema and len(value) > schema["maxLength"]:
            errors.append(f"{label}: 长度须 <= {schema['maxLength']}")

    elif expected == "number" and "multipleOf" in schema:
        factor = Decimal(str(schema["multipleOf"]))
        if (Decimal(str(value)) / factor) % 1 != 0:
            errors.append(f"{label}: 数值须最多两位小数")


if __name__ == "__main__":
    import doctest
    doctest.testmod(verbose=False)
```
