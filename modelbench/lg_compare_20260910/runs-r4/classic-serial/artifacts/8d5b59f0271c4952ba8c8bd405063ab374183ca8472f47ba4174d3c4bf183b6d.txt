```python
"""迷你记账记录校验器：严格依据子任务 A 验收后的 JSON Schema 实现。"""
import re

_TYPE_MAP = {
    "object": dict, "array": list, "string": str,
    "number": (int, float), "integer": int,
    "boolean": bool, "null": type(None),
}


def _type_name(v):
    if v is None: return "null"
    if isinstance(v, bool): return "boolean"
    if isinstance(v, int): return "integer"
    if isinstance(v, float): return "number"
    if isinstance(v, str): return "string"
    if isinstance(v, list): return "array"
    if isinstance(v, dict): return "object"
    return type(v).__name__


def _type_matches(value, t):
    # bool 是 int 的子类，须排除，避免 True 通过 number/integer
    if t in ("number", "integer") and isinstance(value, bool):
        return False
    expected = _TYPE_MAP.get(t)
    if expected is None:
        return False
    if t == "integer":
        return isinstance(value, int)
    return isinstance(value, expected)


def _validate(value, schema, path, errors):
    t = schema.get("type")
    if t is not None:
        allowed = t if isinstance(t, list) else [t]   # 支持联合类型
        if not any(_type_matches(value, x) for x in allowed):
            errors.append(f"{path}: type should be {t}, got {_type_name(value)}")
            return
    if "enum" in schema and value not in schema["enum"]:
        errors.append(f"{path}: value {value} not in enum {schema['enum']}")
    if isinstance(value, str):
        if "minLength" in schema and len(value) < schema["minLength"]:
            errors.append(f"{path}: length {len(value)} < minLength {schema['minLength']}")
        if "maxLength" in schema and len(value) > schema["maxLength"]:
            errors.append(f"{path}: length {len(value)} > maxLength {schema['maxLength']}")
        if "pattern" in schema and not re.search(schema["pattern"], value):
            errors.append(f"{path}: value {value} does not match pattern {schema['pattern']}")
    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in schema and value < schema["minimum"]:
            errors.append(f"{path}: value {value} < minimum {schema['minimum']}")
        if "maximum" in schema and value > schema["maximum"]:
            errors.append(f"{path}: value {value} > maximum {schema['maximum']}")
        if "multipleOf" in schema:
            q = value / schema["multipleOf"]
            if abs(q - round(q)) > 1e-9:   # 容差比较，规避浮点误差
                errors.append(f"{path}: value {value} is not a multiple of {schema['multipleOf']}")
    if isinstance(value, dict):
        props = schema.get("properties", {})
        for key in schema.get("required", []):
            if key not in value:
                errors.append(f"{path}: missing required field {key}")
        if schema.get("additionalProperties") is False:
            for key in value:
                if key not in props:
                    errors.append(f"{path}: unexpected field {key} not allowed (additionalProperties=false)")
        for key, sub in props.items():
            if key in value:
                _validate(value[key], sub, f"{path}.{key}", errors)
    if isinstance(value, list):
        if "items" in schema:
            for i, item in enumerate(value):
                _validate(item, schema["items"], f"{path}[{i}]", errors)
        if schema.get("uniqueItems"):
            seen = []
            for i, item in enumerate(value):
                if item in seen:
                    errors.append(f"{path}[{i}]: duplicate item {item} (uniqueItems=true)")
                else:
                    seen.append(item)


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验一条记账记录，返回违规信息列表（空列表 = 通过）。

    以下 3 个用例中的 schema 字面量均逐字引用子任务 A（wi-03869c8426）的交付。

    >>> schema = {"type": "object", "properties": {"id": {"type": "string", "minLength": 1, "maxLength": 64}, "amount": {"type": "number", "minimum": 0, "multipleOf": 0.01}, "currency": {"type": "string", "minLength": 3, "maxLength": 3, "pattern": "^[A-Z]{3}$"}, "tags": {"type": ["array", "null"], "items": {"type": "string", "minLength": 1}, "uniqueItems": true}}, "required": ["id", "amount", "currency", "tags"], "additionalProperties": false}
    >>> validate_record({"id": "r-001", "amount": 12.5, "currency": "CNY", "tags": ["food", "taxi"]}, schema)
    []
    >>> validate_record({"id": "r-002", "amount": 0.0, "currency": "USD", "tags": None}, schema)
    []
    >>> validate_record({"id": "r-003", "amount": 12.345, "currency": "cny", "tags": [], "note": "extra"}, schema)
    ['$: unexpected field note not allowed (additionalProperties=false)', '$.amount: value 12.345 is not a multiple of 0.01', '$.currency: value cny does not match pattern ^[A-Z]{3}$']
    """
    errors: list[str] = []
    _validate(record, schema, "$", errors)
    return errors


if __name__ == "__main__":
    import doctest
    doctest.testmod(verbose=False)
```
