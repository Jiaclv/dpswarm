def base_convert(s: str, from_base: int, to_base: int) -> str:
    """将字符串 s 从 from_base 进制转换为 to_base 进制（支持 2-36）。

    >>> base_convert("1010", 2, 10)
    '10'
    >>> base_convert("255", 10, 16)
    'FF'
    >>> base_convert("-1A", 36, 2)
    '-1111010'
    """
    if not (2 <= from_base <= 36 and 2 <= to_base <= 36):
        raise ValueError("进制必须在 2-36 之间")
    digits = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    neg = s.startswith("-")
    if neg:
        s = s[1:]
    n = 0
    for ch in s.upper():
        v = digits.find(ch)
        if v < 0 or v >= from_base:
            raise ValueError(f"非法字符: {ch}")
        n = n * from_base + v
    if n == 0:
        return "0"
    out = []
    while n:
        n, r = divmod(n, to_base)
        out.append(digits[r])
    return ("-" if neg else "") + "".join(reversed(out))
