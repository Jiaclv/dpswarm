```python
"""record_validate.py — 记录校验器（纯标准库）。

注意：下方 PLACEHOLDER_SCHEMA 不是子任务 0 的交付文本，仅作临时占位；
拿到原文后须整段替换，并复核关键词覆盖是否与原文严格一致。
"""

from __future__ import annotations

# ⚠️ 占位 schema —— 不是子任务 0 原文，替换前不得对外交付
PLACEHOLDER_SCHEMA = {
    "type": "object",
    "required": ["id", "email"],
    "properties": {
        "id": {"type": "integer", "minimum": 1},
        "email": {"type": "string", "maxLength": 254},
        "role": {"type": "string", "enum": ["admin", "user"]},
        "tags": {"type": "array", "items": {"type": "string"}, "maxItems": 5},
    },
}

_TYPE_CHECKS = {
    "object": lambda v: isinstance(v, dict),
    "array": lambda v: isinstance(v, list),
    "string": lambda v: isinstance(v, str),
    "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
    "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
    "boolean": lambda v: isinstance(v, bool),
    "null": lambda v: v is None,
}


def _type_name(value) -> str:
    if value is None: return "null"
    if isinstance(value, bool): return "boolean"
    if isinstance(value, int): return "integer"
    if isinstance(value, float): return "number"
    if isinstance(value, str): return "string"
    if isinstance(value, list): return "array"
    if isinstance(value, dict): return "object"
    return type(value).__name__


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回错误列表（空列表 = 通过）。

    假设 schema 本身合法（已通过子任务 0 验收）。支持关键词：type（含类型
    数组）、required、properties、items、enum、minimum/maximum、
    minLength/maxLength、minItems/maxItems。错误格式为 "路径: 说明"，根路径
    记作 root；类型不匹配时短路该子树的后续检查；输出顺序遵循 schema 字面
    顺序，结果确定可复现。

    >>> validate_record(
    ...     {"id": 7, "email": "a@b.co", "role": "user", "tags": ["x"]},
    ...     PLACEHOLDER_SCHEMA,
    ... )
    []

    >>> for err in validate_record(
    ...     {"id": 0, "role": "ghost", "tags": [1]},
    ...     PLACEHOLDER_SCHEMA,
    ... ):
    ...     print(err)
    root: missing required property 'email'
    id: expected >= 1, got 0
    role: value 'ghost' not in enum ['admin', 'user']
    tags[0]: expected type string, got int

    >>> validate_record({"id": "x", "email": "a@b.co"}, PLACEHOLDER_SCHEMA)
    ['id: expected type integer, got str']
    """
    return _check(record, schema, path="")


def _check(value, schema, path: str) -> list[str]:
    label = path or "root"
    errors: list[str] = []

    t = schema.get("type")
    if t is not None:
        types = t if isinstance(t, list) else [t]
        if not any(_TYPE_CHECKS[k](value) for k in types):
            return [f"{label}: expected type {t}, got {_type_name(value)}"]

    if "enum" in schema and value not in schema["enum"]:
        errors.append(f"{label}: value {value!r} not in enum {schema['enum']!r}")

    if isinstance(value, dict):
        for key in schema.get("required", ()):
            if key not in value:
                errors.append(f"{label}: missing required property '{key}'")
        for key, sub in schema.get("properties", {}).items():
            if key in value:
                errors.extend(_check(value[key], sub, f"{path}.{key}" if path else key))
    elif isinstance(value, list):
        n = len(value)
        if "minItems" in schema and n < schema["minItems"]:
            errors.append(f"{label}: expected >= {schema['minItems']} items, got {n}")
        if "maxItems" in schema and n > schema["maxItems"]:
            errors.append(f"{label}: expected <= {schema['maxItems']} items, got {n}")
        if "items" in schema:
            for i, elem in enumerate(value):
                errors.extend(_check(elem, schema["items"], f"{path}[{i}]"))
    elif isinstance(value, str):
        n = len(value)
        if "minLength" in schema and n < schema["minLength"]:
            errors.append(f"{label}: expected length >= {schema['minLength']}, got {n}")
        if "maxLength" in schema and n > schema["maxLength"]:
            errors.append(f"{label}: expected length <= {schema['maxLength']}, got {n}")
    elif isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in schema and value < schema["minimum"]:
            errors.append(f"{label}: expected >= {schema['minimum']}, got {value}")
        if "maximum" in schema and value > schema["maximum"]:
            errors.append(f"{label}: expected <= {schema['maximum']}, got {value}")

    return errors

    import doctest
    doctest.testmod()
```
