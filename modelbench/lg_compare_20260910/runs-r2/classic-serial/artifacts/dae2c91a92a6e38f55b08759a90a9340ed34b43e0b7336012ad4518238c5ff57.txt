```text
{
  "type": "object",
  "required": ["id", "name", "tags"],
  "properties": {
    "id":   {"type": "integer", "minimum": 1},
    "name": {"type": "string", "minLength": 3, "maxLength": 32},
    "tags": {"type": "array", "items": {"type": "string", "minLength": 1}},
    "role": {"type": "string", "enum": ["admin", "user", "guest"]}
  },
  "additionalProperties": false

```

## 完整实现

```python
"""validate_record —— 纯标准库记录校验器。

schema 文本逐字来源于子任务 0 验收产物（pkg://wi-9cf6e99b23-r1/65d5159d52fc）。
返回错误列表（list[str]），空列表表示校验通过。
"""

_TYPES = {
    "object":  lambda v: isinstance(v, dict),
    "array":   lambda v: isinstance(v, list),
    "string":  lambda v: isinstance(v, str),
    "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
    "number":  lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
    "boolean": lambda v: isinstance(v, bool),
    "null":    lambda v: v is None,
}


def _type_ok(value, expected):
    """类型分发表：单个类型名或类型名列表（联合类型）。"""
    if isinstance(expected, list):
        return any(_type_ok(value, t) for t in expected)
    if expected in _TYPES:
        return _TYPES[expected](value)
    return True  # 未知类型约束不生效，避免误报


def _fail(errors, path, msg):
    errors.append(f"{path}: {msg}")


def _check(value, rule, path, errors):
    if "type" in rule and not _type_ok(value, rule["type"]):
        _fail(errors, path,
              f"expected type {rule['type']!r}, got {type(value).__name__!r}")
        return  # 短路：类型不符时后续约束（minLength 等）无意义，防止误报
    if "enum" in rule and value not in rule["enum"]:
        _fail(errors, path, f"value {value!r} not in enum {rule['enum']!r}")
    if isinstance(value, str):
        if "minLength" in rule and len(value) < rule["minLength"]:
            _fail(errors, path,
                  f"shorter than minLength {rule['minLength']} (len={len(value)})")
        if "maxLength" in rule and len(value) > rule["maxLength"]:
            _fail(errors, path,
                  f"longer than maxLength {rule['maxLength']} (len={len(value)})")
    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in rule and value < rule["minimum"]:
            _fail(errors, path, f"below minimum {rule['minimum']}")
        if "maximum" in rule and value > rule["maximum"]:
            _fail(errors, path, f"above maximum {rule['maximum']}")
    if isinstance(value, list) and "items" in rule:
        for i, item in enumerate(value):
            _check(item, rule["items"], f"{path}[{i}]", errors)
    if isinstance(value, dict):
        for key in rule.get("required", []):
            if key not in value:
                _fail(errors, f"{path}.{key}", "required field missing")
        props = rule.get("properties", {})
        for key, sub in props.items():
            if key in value:
                _check(value[key], sub, f"{path}.{key}", errors)
        if rule.get("additionalProperties") is False:
            for key in value:
                if key not in props:
                    _fail(errors, f"{path}.{key}", "additional property not allowed")


def validate_record(record: dict, schema: dict) -> list[str]:
    """依据 schema 校验 record；返回错误列表，空列表 = 通过。

    >>> schema = {
    ...     "type": "object",
    ...     "required": ["id", "name", "tags"],
    ...     "properties": {
    ...         "id":   {"type": "integer", "minimum": 1},
    ...         "name": {"type": "string", "minLength": 3, "maxLength": 32},
    ...         "tags": {"type": "array", "items": {"type": "string", "minLength": 1}},
    ...         "role": {"type": "string", "enum": ["admin", "user", "guest"]},
    ...     },
    ...     "additionalProperties": False,
    ... }

    用例 1：合法记录 → 空列表（通过）
    >>> validate_record({"id": 7, "name": "alice", "tags": ["a", "b"], "role": "user"}, schema)
    []

    用例 2：多处违规 → 逐一报出（min/max、minLength、enum、多余字段）
    >>> validate_record({"id": 0, "name": "ab", "tags": [""], "role": "root", "extra": 1}, schema)
    ['$.id: below minimum 1', '$.name: shorter than minLength 3 (len=2)', '$.tags[0]: shorter than minLength 1 (len=0)', "$.role: value 'root' not in enum ['admin', 'user', 'guest']", '$.extra: additional property not allowed']

    用例 3：缺必填字段 + 类型不符（短路：类型错误时不附带 minLength 误报）
    >>> validate_record({"name": 42}, schema)
    ['$.id: required field missing', '$.tags: required field missing', "$.name: expected type 'string', got 'int'"]
    """
    errors: list[str] = []
    _check(record, schema, "$", errors)
    return errors


if __name__ == "__main__":
    import doctest
    failures, total = doctest.testmod()
    print(f"doctest: {total - failures}/{total} passed")
```
