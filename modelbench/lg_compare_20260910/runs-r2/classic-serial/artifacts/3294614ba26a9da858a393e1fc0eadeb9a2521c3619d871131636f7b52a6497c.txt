```python
def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表=通过）。

    doctest 中的 SCHEMA 逐字引用自子任务 A 验收文本（pkg://wi-64a345dd00-r1/c0ebaae8f008）。

    >>> SCHEMA = {
    ...     "type": "object",
    ...     "required": ["id", "name"],
    ...     "properties": {
    ...         "id": {"type": "integer"},
    ...         "name": {"type": "string", "minLength": 1},
    ...         "active": {"type": "boolean"},
    ...         "tags": {"type": "array", "items": {"type": "string"}},
    ...         "children": {"type": "array", "items": {"$ref": "#"}}
    ...     }
    ... }
    >>> validate_record({"id": 1, "name": "a", "active": True, "tags": ["x"],
    ...                  "children": [{"id": 2, "name": "b"}]}, SCHEMA)
    []
    >>> validate_record({"id": True, "name": ""}, SCHEMA)
    ['$.id: expected integer, got bool', '$.name: length 0 < minLength 1']
    >>> validate_record({"id": 1, "name": "a", "children": [{"id": "x"}]}, SCHEMA)
    ['$.children[0].id: expected integer, got str', "$.children[0]: missing required property 'name'"]
    """
    root = schema  # $ref "#" 的解析目标

    def resolve(s):
        if isinstance(s, dict) and s.get("$ref") == "#":
            return root
        return s

    def check(value, s, path, errors):
        s = resolve(s)
        t = s.get("type")
        if t == "object":
            if not isinstance(value, dict):
                errors.append(f"{path}: expected object, got {type(value).__name__}")
                return
            for req in s.get("required", []):
                if req not in value:
                    errors.append(f"{path}: missing required property {req!r}")
            for key, sub in s.get("properties", {}).items():
                if key in value:
                    check(value[key], sub, f"{path}.{key}", errors)
        elif t == "array":
            if not isinstance(value, list):
                errors.append(f"{path}: expected array, got {type(value).__name__}")
                return
            for i, item in enumerate(value):
                check(item, s.get("items", {}), f"{path}[{i}]", errors)
        elif t == "string":
            if not isinstance(value, str):
                errors.append(f"{path}: expected string, got {type(value).__name__}")
                return
            if "minLength" in s and len(value) < s["minLength"]:
                errors.append(f"{path}: length {len(value)} < minLength {s['minLength']}")
        elif t == "integer":
            if isinstance(value, bool) or not isinstance(value, int):
                errors.append(f"{path}: expected integer, got {type(value).__name__}")
        elif t == "boolean":
            if not isinstance(value, bool):
                errors.append(f"{path}: expected boolean, got {type(value).__name__}")

    errors: list[str] = []
    check(record, schema, "$", errors)
    return errors
```

**实现层裁量决策（按打回指示自行决策并注明）**：

1. **违规信息模板**：采用 `"<JSON路径>: <描述>"` 格式（如 `$.children[0].id: expected integer, got str`），路径可精确定位嵌套违规点，满足「返回 list[str]、空列表=通过」的全部任务约束。
2. **嵌套递归规则**：`$ref: "#"` 解析为传入的根 schema（闭包内 `root` 引用），`children` 任意深度递归校验，每个嵌套层独立执行 required/type 检查（doctest 第 3 例已覆盖子节点缺失 `name` 的情形）。
3. **bool/int 边界**：依据 Python 中 `bool` 是 `int` 子类的语言事实，在 integer 分支显式排除 `isinstance(value, bool)`（doctest 第 2 例覆盖：`"id": True` 判违规并报 `got bool`）；boolean 分支仅接受真正的 `bool`。
