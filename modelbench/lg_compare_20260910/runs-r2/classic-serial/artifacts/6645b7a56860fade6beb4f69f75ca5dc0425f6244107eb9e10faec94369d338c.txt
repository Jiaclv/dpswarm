import json


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回错误信息列表；空列表表示通过。

    当前按“type + required”最小约定实现；收到正式 schema 后按其逐字语义调整。

    >>> validate_record({"id": 1, "name": "a"}, json.loads(SCHEMA_TEXT))
    []

    ["field 'id': expected integer, got str"]
    >>> validate_record({"id": 1}, json.loads(SCHEMA_TEXT))
    ["missing required field 'name'"]
    """
    errors: list[str] = []
    for field in schema.get("required", []):
        if field not in record:
            errors.append(f"missing required field '{field}'")
    for field, rule in schema.get("properties", {}).items():
        if field in record and rule.get("type") == "string" and not isinstance(record[field], str):
            errors.append(f"field '{field}': expected string, got {type(record[field]).__name__}")
        elif field in record and rule.get("type") == "integer" and not isinstance(record[field], int):
            errors.append(f"field '{field}': expected integer, got {type(record[field]).__name__}")
    return errors
