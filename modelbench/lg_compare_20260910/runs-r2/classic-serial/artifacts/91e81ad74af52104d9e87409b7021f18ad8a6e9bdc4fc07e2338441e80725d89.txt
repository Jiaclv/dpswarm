def validate_record(record: dict, schema: dict) -> list[str]:
    """Validate record against schema; empty list means pass.

    Schema type is jsonschema-like; standard library only.

    >>> schema = {
    ...     "type": "object",
    ...     "required": ["id", "name"],
    ...     "properties": {
    ...         "id": {"type": "integer"},
    ...         "name": {"type": "string"},
    ...         "tags": {"type": "array", "items": {"type": "string"}, "minItems": 1, "maxItems": 3}
    ...     }
    ... }
    >>> validate_record({"id": "x", "name": "n", "tags": []}, schema)
    ['id: expected integer, got str', 'tags: expected at least 1 item, got 0']
    >>> validate_record({"id": 1, "name": "a", "tags": ["x", 2, "y", "z"]}, schema)
    ['tags[1]: expected string, got int', 'tags: expected at most 3 items, got 4']
    >>> validate_record({"id": 1, "name": "a", "tags": ["x"]}, schema)
    []
    """
    errors: list[str] = []
    if schema.get("type") == "object":
        for key in schema.get("required", []):
            if key not in record:
                errors.append(f"{key}: missing required field")
        for key, sub in schema.get("properties", {}).items():
            if key in record:
                errors.extend(_validate_value(key, record[key], sub))
    return errors


def _validate_value(path: str, value, spec: dict) -> list[str]:
    errors: list[str] = []
    expected = spec.get("type")
    if expected and not _type_ok(value, expected):
        errors.append(f"{path}: expected {expected}, got {type(value).__name__}")
        return errors
    if expected == "array":
        items_spec = spec.get("items", {})
        for i, item in enumerate(value):
            errors.extend(_validate_value(f"{path}[{i}]", item, items_spec))
        if "minItems" in spec and len(value) < spec["minItems"]:
            errors.append(f"{path}: expected at least {spec['minItems']} items, got {len(value)}")
        if "maxItems" in spec and len(value) > spec["maxItems"]:
            errors.append(f"{path}: expected at most {spec['maxItems']} items, got {len(value)}")
    return errors


def _type_ok(value, expected: str) -> bool:
    if expected == "integer":
        # bool 是 int 的子类，显式排除
        return isinstance(value, int) and not isinstance(value, bool)
    if expected == "string":
        return isinstance(value, str)
    if expected == "array":
        return isinstance(value, list)
    return True
