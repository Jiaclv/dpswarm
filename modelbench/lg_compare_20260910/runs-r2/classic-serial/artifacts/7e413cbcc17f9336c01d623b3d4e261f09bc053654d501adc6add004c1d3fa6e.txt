def validate_record(record: dict, schema: dict) -> list[str]:
    """依据 schema 校验 record，返回错误列表（空列表 = 通过）。

    支持的关键字：type / required / properties / enum / items / min / max。

    >>> SCHEMA = {  # TODO: 逐字替换为子任务 0 的交付文本
    ...     "type": "object",
    ...     "required": ["id", "score"],
    ...     "properties": {"id": {"type": "integer"}, "score": {"type": "number"}},
    ... }
    >>> validate_record({"id": 1, "score": 9.5}, SCHEMA)
    []
    >>> validate_record({"id": "x", "score": 9.5}, SCHEMA)
    ["id: expected integer, got str"]
    >>> validate_record({"id": 1}, SCHEMA)
    ["score: missing required field"]
    """
    errors: list[str] = []

    def check(value, spec, path):
        t = spec.get("type")
        type_map = {"object": dict, "array": list, "string": str,
                    "integer": int, "number": (int, float), "boolean": bool}
        if t and not (isinstance(value, type_map[t])
                      and not (t in ("integer", "number") and isinstance(value, bool))):
            errors.append(f"{path or 'root'}: expected {t}, got {type(value).__name__}")
            return
        if "enum" in spec and value not in spec["enum"]:
            errors.append(f"{path}: value {value!r} not in enum {spec['enum']}")
        if isinstance(value, dict) and spec.get("type") == "object":
            for key in spec.get("required", []):
                if key not in value:
                    errors.append(f"{path or 'root'}: missing required field {key!r}")
            for key, sub in spec.get("properties", {}).items():
                if key in value:
                    check(value[key], sub, f"{path}.{key}" if path else key)
        if isinstance(value, list) and spec.get("type") == "array" and "items" in spec:
            for i, item in enumerate(value):
                check(item, spec["items"], f"{path}[{i}]")

    check(record, schema, "")
    return errors
