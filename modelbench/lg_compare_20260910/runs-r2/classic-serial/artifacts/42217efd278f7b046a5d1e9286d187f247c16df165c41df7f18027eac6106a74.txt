# validator.py —— 纯标准库实现（仅 re / json）
import json
import re


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回错误列表（空列表 = 通过）。

    支持关键字子集：type, required, properties, enum,
    minimum, maximum, minLength, maxLength, pattern, items。

    >>> A_SCHEMA_TEXT = '''
    ... {
    ...   "type": "object",
    ...   "required": ["id", "email"],
    ...   "properties": {
    ...     "id":     {"type": "integer", "minimum": 1},
    ...     "email":  {"type": "string", "pattern": "^[^@]+@[^@]+$"},
    ...     "role":   {"type": "string", "enum": ["admin", "user"]},
    ...     "tags":   {"type": "array", "items": {"type": "string"}}
    ...   }
    ... }
    ... '''
    >>> schema = json.loads(A_SCHEMA_TEXT)

    合法记录 → 空列表
    >>> validate_record({"id": 7, "email": "a@b.c", "role": "user", "tags": ["x"]}, schema)
    []

    缺失必填字段 + id 低于 minimum
    >>> validate_record({"id": 0}, schema)
    ["missing required field: 'email'", "id: 0 < minimum 1"]

    email 不匹配 pattern + role 不在 enum
    >>> validate_record({"id": 1, "email": "bad", "role": "root"}, schema)
    ["email: does not match pattern '^[^@]+@[^@]+$'", "role: 'root' not in enum ['admin', 'user']"]
    """
    errors: list[str] = []

    if schema.get("type") == "object":
        for req in schema.get("required", []):
            if req not in record:
                errors.append(f"missing required field: '{req}'")
        props = schema.get("properties", {})
        for key, sub in props.items():
            if key in record:
                errors.extend(validate_record(record[key], sub) and
                              [f"{key}: {e}" for e in validate_record(record[key], sub)]
                              or [])
    return errors or _check_leaf(record, schema)


def _check_leaf(value, schema: dict) -> list[str]:
    errors: list[str] = []
    t = schema.get("type")
    type_map = {"string": str, "integer": int, "number": (int, float),
                "boolean": bool, "array": list, "object": dict, "null": type(None)}
    if t and not (isinstance(value, type_map[t]) and not (t in ("integer", "number") and isinstance(value, bool))):
        errors.append(f"expected type '{t}', got {type(value).__name__}")
        return errors
    if "enum" in schema and value not in schema["enum"]:
        errors.append(f"not in enum {schema['enum']!r}")
    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in schema and value < schema["minimum"]:
            errors.append(f"{value} < minimum {schema['minimum']}")
        if "maximum" in schema and value > schema["maximum"]:
            errors.append(f"{value} > maximum {schema['maximum']}")
    if isinstance(value, str):
        if "minLength" in schema and len(value) < schema["minLength"]:
            errors.append(f"length {len(value)} < minLength {schema['minLength']}")
        if "maxLength" in schema and len(value) > schema["maxLength"]:
            errors.append(f"length {len(value)} > maxLength {schema['maxLength']}")
        if "pattern" in schema and not re.search(schema["pattern"], value):
            errors.append(f"does not match pattern {schema['pattern']!r}")
    if isinstance(value, list) and "items" in schema:
        for i, item in enumerate(value):
            for e in validate_record(item, schema["items"]):
                errors.append(f"[{i}]: {e}")
    return errors
