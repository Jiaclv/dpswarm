```python
def validate_record(record: dict, schema: dict) -> list[str]:
    """依据 schema 校验 record，返回违规信息列表（空列表 = 通过）。

    假设 schema 语义为（以 A 的实际文本为准调整）：
      - "type": record 顶层类型
      - "required": 必填字段名列表
      - "properties": {字段名: {"type": 类型, "enum": 可选取值(可选)}}

    >>> validate_record({"id": 1, "name": "x"}, {"type": "object", "required": ["id", "name"], "properties": {"id": {"type": "integer"}, "name": {"type": "string"}}})
    []
    >>> validate_record({"id": "1", "name": "x"}, {"type": "object", "required": ["id", "name"], "properties": {"id": {"type": "integer"}, "name": {"type": "string"}}})
    ["字段 'id' 类型错误：期望 integer，实际 str"]
    >>> validate_record({"name": "x"}, {"type": "object", "required": ["id", "name"], "properties": {"id": {"type": "integer"}, "name": {"type": "string"}}})
    ["缺少必填字段：'id'"]
    """
    errors: list[str] = []

    if schema.get("type") == "object" and not isinstance(record, dict):
        return ["顶层类型错误：期望 object"]

    for field in schema.get("required", []):
        if field not in record:
            errors.append(f"缺少必填字段：'{field}'")

    for field, rule in schema.get("properties", {}).items():
        if field in record:
            expected = rule.get("type")
            actual = type(record[field]).__name__
            if expected == "integer" and not isinstance(record[field], int):
                errors.append(f"字段 '{field}' 类型错误：期望 {expected}，实际 {actual}")
            elif expected == "string" and not isinstance(record[field], str):
                errors.append(f"字段 '{field}' 类型错误：期望 {expected}，实际 {actual}")
            if "enum" in rule and record[field] not in rule["enum"]:
                errors.append(f"字段 '{field}' 取值不在允许范围内")

    return errors
```
