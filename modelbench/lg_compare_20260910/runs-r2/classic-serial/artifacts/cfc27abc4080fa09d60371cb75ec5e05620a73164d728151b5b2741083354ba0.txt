# -*- coding: utf-8 -*-
"""validate_record —— 依据 A 验收后的 schema 实现的记录校验器。

纯标准库实现，零 import。

# ---- 以下 schema 逐字引自 A 的验收交付（pkg://wi-61d4012b8b-r1/1d54f130f79c），不得改动 ----
SCHEMA = {
    "fields": {
        "id": {"type": "integer", "required": True},
        "name": {"type": "string", "required": True},
        "role": {"type": "string", "required": True,
                 "enum": ["admin", "user"]},
    }
}
# ---- A 的 schema 原文到此为止 ----


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验 record 是否符合 schema，返回违规信息列表（空列表 = 通过）。

    >>> validate_record({"id": 1, "name": "alice", "role": "admin"}, SCHEMA)
    []
    >>> validate_record({"id": 1, "role": "user"}, SCHEMA)
    ["字段 'name' 缺失"]
    >>> validate_record({"id": "x", "name": "bob", "role": "guest"}, SCHEMA)
    ["字段 'id' 类型错误：期望 integer，实际 str", "字段 'role' 取值不在枚举范围内：guest"]
    """
    violations: list[str] = []
    fields = schema["fields"]

    for name, spec in fields.items():
        if name not in record:
            if spec.get("required", False):
                violations.append(f"字段 '{name}' 缺失")
            continue

        value = record[name]
        expected = spec["type"]

        # 注意：isinstance(True, int) 为真，需显式排除 bool 冒充 integer
        if expected == "integer":
            ok = isinstance(value, int) and not isinstance(value, bool)
        elif expected == "string":
            ok = isinstance(value, str)
        else:
            ok = isinstance(value, expected)

        if not ok:
            actual = type(value).__name__
            violations.append(f"字段 '{name}' 类型错误：期望 {expected}，实际 {actual}")
            continue

        if "enum" in spec and value not in spec["enum"]:
            violations.append(f"字段 '{name}' 取值不在枚举范围内：{value}")

    return violations
