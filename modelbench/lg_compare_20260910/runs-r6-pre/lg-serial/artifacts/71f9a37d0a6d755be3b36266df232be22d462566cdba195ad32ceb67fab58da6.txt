```python
"""validate_record：纯标准库 JSON Schema 子集校验器（迷你记账记录）。"""

import json
import re
from decimal import Decimal

# ==== 逐字引用子任务 A 验收交付的 schema 全文（仅此一处引用源）====
SCHEMA_A_TEXT = r'''
{
  "$id": "https://example.com/mini-ledger/record.schema.json",
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "title": "迷你记账记录",
  "description": "一条迷你记账记录：唯一 id、两位小数金额、三字母货币代码与可空标签数组",
  "type": "object",
  "properties": {
    "id": {
      "description": "记录唯一标识，非空字符串",
      "type": "string",
      "minLength": 1
    },
    "amount": {
      "description": "金额，保留两位小数，不小于 0",
      "type": "number",
      "minimum": 0,
      "multipleOf": 0.01
    },
    "currency": {
      "description": "ISO 4217 三字母货币代码",
      "type": "string",
      "pattern": "^[A-Z]{3}$"
    },
    "tags": {
      "description": "标签数组，可为空数组，元素不重复",
      "type": "array",
      "items": {
        "type": "string"
      },
      "uniqueItems": true
    }
  },
  "required": ["id", "amount", "currency"],
  "additionalProperties": false
}
'''.strip()

# doctest 统一使用此常量，保证与 A 的交付文本严格一致
LEDGER_SCHEMA = json.loads(SCHEMA_A_TEXT)


def _fmt(path: str, msg: str) -> str:
    return f"{path}: {msg}" if path else msg


def _type_ok(value, expected: str) -> bool:
    if expected == "string":
        return isinstance(value, str)
    if expected == "number":
        return isinstance(value, (int, float)) and not isinstance(value, bool)
    if expected == "integer":
        return isinstance(value, int) and not isinstance(value, bool)
    if expected == "boolean":
        return isinstance(value, bool)
    if expected == "array":
        return isinstance(value, list)
    if expected == "object":
        return isinstance(value, dict)
    return True


def _check(value, schema: dict, path: str) -> list[str]:
    errors: list[str] = []
    expected = schema.get("type")
    if expected and not _type_ok(value, expected):
        return [_fmt(path, f"类型应为 {expected}，实际为 {type(value).__name__}")]
    if isinstance(value, str):
        if "minLength" in schema and len(value) < schema["minLength"]:
            errors.append(_fmt(path, f"长度 {len(value)} 小于 minLength {schema['minLength']}"))
        if "maxLength" in schema and len(value) > schema["maxLength"]:
            errors.append(_fmt(path, f"长度 {len(value)} 超过 maxLength {schema['maxLength']}"))
        if "pattern" in schema and not re.search(schema["pattern"], value):
            errors.append(_fmt(path, f"不匹配 pattern {schema['pattern']}"))
    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in schema and value < schema["minimum"]:
            errors.append(_fmt(path, f"{value} 小于 minimum {schema['minimum']}"))
        if "maximum" in schema and value > schema["maximum"]:
            errors.append(_fmt(path, f"{value} 超过 maximum {schema['maximum']}"))
        if "multipleOf" in schema:
            if Decimal(str(value)) % Decimal(str(schema["multipleOf"])) != 0:
                errors.append(_fmt(path, f"{value} 不是 {schema['multipleOf']} 的整数倍"))
    if isinstance(value, list):
        if schema.get("uniqueItems"):
            seen: list = []
            for item in value:
                if item in seen:
                    errors.append(_fmt(path, f"存在重复元素 {item!r}"))
                    break
                seen.append(item)
        items = schema.get("items")
        if items:
            for i, item in enumerate(value):
                errors.extend(_check(item, items, f"{path}[{i}]"))
    if isinstance(value, dict):
        for key in schema.get("required", []):
            if key not in value:
                errors.append(_fmt(path, f"缺少必填字段 {key}"))
        props = schema.get("properties", {})
        for key, sub in props.items():
            if key in value:
                errors.extend(_check(value[key], sub, key if not path else f"{path}.{key}"))
        if schema.get("additionalProperties") is False:
            for key in value:
                if key not in props:
                    errors.append(_fmt(path, f"多余字段 {key}"))
    return errors


def validate_record(record: dict, schema: dict) -> list[str]:
    """按 schema 校验 record，返回违规信息列表（空列表表示通过）。

    支持：type / properties / required / additionalProperties /
    minLength / maxLength / pattern / minimum / maximum /
    multipleOf（用 Decimal 规避浮点误差）/ items / uniqueItems。

    >>> validate_record({"id": "r001", "amount": 12.5, "currency": "CNY", "tags": []}, LEDGER_SCHEMA)
    []
    >>> validate_record({"id": "", "amount": -3.456, "currency": "c1", "tags": ["food", "food"], "note": "x"}, LEDGER_SCHEMA)
    ['id: 长度 0 小于 minLength 1', 'amount: -3.456 小于 minimum 0', 'amount: -3.456 不是 0.01 的整数倍', 'currency: 不匹配 pattern ^[A-Z]{3}$', "tags: 存在重复元素 'food'", 'note: 多余字段 note']
    >>> validate_record({"id": "r002", "amount": 1.0, "currency": "USD", "tags": "urgent"}, LEDGER_SCHEMA)
    ['tags: 类型应为 array，实际为 str']
    """
    return _check(record, schema, "")


if __name__ == "__main__":
    import doctest
    print(doctest.testmod(verbose=False))
```
