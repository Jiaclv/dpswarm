```python
"""迷你记账记录校验器（纯标准库实现）。

SCHEMA 逐字引用自子任务 A 已验收交付文本
（context: pkg://wi-aec703dfe0-r1/c2c8a1e2de58）。
"""
import re

# ---- 以下 SCHEMA 为子任务 A 已验收交付的逐字内嵌，未作任何改动 ----
SCHEMA = {
    "$schema": "https://json-schema.org/draft/2020-12/schema",
    "title": "迷你记账记录",
    "type": "object",
    "additionalProperties": False,
    "required": ["id", "amount", "currency", "tags"],
    "properties": {
        "id": {"type": "string"},
        "amount": {"type": "number", "multipleOf": 0.01},
        "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
        "tags": {"type": ["array", "null"], "items": {"type": "string"}}
    }
}
# ---- 逐字内嵌结束 ----


def validate_record(record: dict, schema: dict) -> list[str]:
    """校验单条迷你记账记录，返回违规信息列表；空列表表示通过。

    >>> validate_record({"id": "r-001", "amount": 12.34, "currency": "CNY", "tags": ["早餐", "通勤"]}, SCHEMA)
    []
    >>> validate_record({"id": "r-002", "amount": 1.999, "currency": "usd", "tags": [1, "打车"]}, SCHEMA)
    ['字段「amount」必须为两位小数', '字段「currency」必须为三字母大写代码（^[A-Z]{3}$）', '字段「tags」的第 1 个元素不是字符串']
    >>> validate_record({"id": 7, "amount": "12.00", "currency": "JPY", "tags": None}, SCHEMA)
    ['字段「id」不是字符串', '字段「amount」不是数值']
    """
    errors: list[str] = []
    props = schema["properties"]

    for name in schema.get("required", []):
        if name not in record:
            errors.append(f"缺少必填字段「{name}」")
    for name in record:
        if name not in props:
            errors.append(f"未知字段「{name}」")

    def is_number(v):
        return isinstance(v, (int, float)) and not isinstance(v, bool)

    v = record.get("id")
    if v is not None and not isinstance(v, str):
        errors.append("字段「id」不是字符串")

    v = record.get("amount")
    if v is not None and not is_number(v):
        errors.append("字段「amount」不是数值")
    elif is_number(v) and round(v, 2) != v:
        errors.append("字段「amount」必须为两位小数")

    v = record.get("currency")
    if v is not None and not (isinstance(v, str) and re.fullmatch(r"[A-Z]{3}", v)):
        errors.append("字段「currency」必须为三字母大写代码（^[A-Z]{3}$）")

    v = record.get("tags")
    if v is not None:
        if not isinstance(v, list):
            errors.append("字段「tags」不是字符串数组")
        else:
            for i, item in enumerate(v, 1):
                if not isinstance(item, str):
                    errors.append(f"字段「tags」的第 {i} 个元素不是字符串")

    return errors
```
