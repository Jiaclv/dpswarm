def parse_kv(text):
    """将 key=value 文本解析为 dict。

    每行一个键值对，以首个 '=' 分隔；空行跳过。

    >>> parse_kv("host=localhost\\nport=8080")
    {'host': 'localhost', 'port': '8080'}

    >>> parse_kv("token=a=b\\n\\nflag=on")
    {'token': 'a=b', 'flag': 'on'}
    """
    result = {}
    for line in text.splitlines():
        line = line.strip()
        if not line or "=" not in line:
            continue
        key, _, value = line.partition("=")
        result[key.strip()] = value.strip()
    return result
