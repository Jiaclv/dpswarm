```python
"""迷你记账记录校验器：validate_record(record, schema) -> list[str]。空列表 = 通过。"""

import re

_TYPE_CHECKS = {
    "string": lambda v: isinstance(v, str),
    "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
    "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
    "boolean": lambda v: isinstance(v, bool),
    "array": lambda v: isinstance(v, list),
    "object": lambda v: isinstance(v, dict),
    "null": lambda v: v is None,
}


def validate_record(record: dict, schema: dict) -> list[str]:
    """按 schema 校验单条记录，返回违规信息列表（空列表 = 通过）。

    >>> schema = {
    ...     "$schema": "http://json-schema.org/draft-07/schema#",
    ...     "type": "object",
    ...     "properties": {
    ...         "id": {"type": "string", "minLength": 1},
    ...         "amount": {"type": "number", "exclusiveMinimum": 0, "multipleOf": 0.01},
    ...         "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
    ...         "tags": {"type": ["array", "null"], "items": {"type": "string"}},
    ...     },
    ...     "required": ["id", "amount", "currency", "tags"],
    ...     "additionalProperties": False,
    ... }
    >>> validate_record({"id": "r1", "amount": 12.34, "currency": "CNY", "tags": ["food", "taxi"]}, schema)
    []

    >>> validate_record({"id": "", "amount": 0, "currency": "cny", "tags": [1]}, schema)
    ['minLength: 长度 0 < 1', 'exclusiveMinimum: 0 <= 0', "pattern: 'cny' 不匹配 ^[A-Z]{3}$", 'items[0].type: 期望 string, 实得 int']

    >>> validate_record({"id": "r2", "amount": 1.234, "currency": "USD", "tags": None, "note": "x"}, schema)
    ['multipleOf: 1.234 不是 0.01 的整数倍', 'additionalProperties: 多余字段 note']
    """
    errors: list[str] = []

    # —— type（支持联合类型分派，如 ["array", "null"]）——
    if "type" in schema:
        types = schema["type"] if isinstance(schema["type"], list) else [schema["type"]]
        if not any(_TYPE_CHECKS[t](record) for t in types):
            errors.append(f"type: 期望 {'/'.join(types)}, 实得 {type(record).__name__}")
            return errors  # 类型不符时不再深入逐条检查

    # —— required ——
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"required: 缺少字段 {key}")

    # —— properties（递归）——
    for key, sub in schema.get("properties", {}).items():
        if key in record:
            errors.extend(validate_record(record[key], sub))

    # —— 字符串约束 ——
    if isinstance(record, str):
        if "minLength" in schema and len(record) < schema["minLength"]:
            errors.append(f"minLength: 长度 {len(record)} < {schema['minLength']}")
        if "pattern" in schema and not re.search(schema["pattern"], record):
            errors.append(f"pattern: {record!r} 不匹配 {schema['pattern']}")

    # —— 数值约束 ——
    if isinstance(record, (int, float)) and not isinstance(record, bool):
        if "exclusiveMinimum" in schema and record <= schema["exclusiveMinimum"]:
            errors.append(f"exclusiveMinimum: {record} <= {schema['exclusiveMinimum']}")
        if "multipleOf" in schema:
            q = record / schema["multipleOf"]
            if abs(q - round(q)) > 1e-9:  # 容差规避浮点误差
                errors.append(f"multipleOf: {record} 不是 {schema['multipleOf']} 的整数倍")

    # —— 数组约束 ——
    if isinstance(record, list) and "items" in schema:
        for i, item in enumerate(record):
            errors.extend(f"items[{i}].{e}" for e in validate_record(item, schema["items"]))

    # —— 对象约束 ——
    if isinstance(record, dict) and schema.get("additionalProperties") is False:
        for key in record:
            if key not in schema.get("properties", {}):
                errors.append(f"additionalProperties: 多余字段 {key}")

    return errors


if __name__ == "__main__":
    import doctest
    doctest.testmod(verbose=False)
```
