```python
"""Validator for the mini-ledger record schema (subtask B, deps=[0]).

Standard library only. Supports exactly the keyword set used by the
accepted schema: type, properties, required, additionalProperties,
minLength, pattern, multipleOf, items, minItems.
Requires Python >= 3.9.
"""
import re
from decimal import Decimal

_JSON_TYPE_NAMES = {
    type(None): "null",
    bool: "boolean",   # must precede int: bool is a subclass of int
    int: "integer",
    float: "number",
    str: "string",
    list: "array",
    dict: "object",
}


def _json_type(value):
    for t, name in _JSON_TYPE_NAMES.items():
        if isinstance(value, t):
            return name
    return type(value).__name__


def _type_matches(value, expected):
    actual = _json_type(value)
    if expected == "number":
        return actual in ("integer", "number")
    return actual == expected


def _validate_value(path, value, subschema):
    errors = []
    expected = subschema.get("type")
    if expected is not None and not _type_matches(value, expected):
        errors.append(f"{path}: expected {expected}, got {_json_type(value)}")
        return errors  # skip remaining constraints on a type mismatch

    if "minLength" in subschema and len(value) < subschema["minLength"]:
        errors.append(f"{path}: string length {len(value)} is less than {subschema['minLength']}")

    if "pattern" in subschema and re.search(subschema["pattern"], value) is None:
        errors.append(f"{path}: {value!r} does not match pattern {subschema['pattern']!r}")

    if "multipleOf" in subschema:
        quant = Decimal(str(subschema["multipleOf"]))
        num = Decimal(str(value))
        if quant == 0 or (num / quant) % 1 != 0:
            errors.append(f"{path}: {value!r} is not a multiple of {subschema['multipleOf']!r}")

    if "minItems" in subschema and len(value) < subschema["minItems"]:
        errors.append(f"{path}: array length {len(value)} is less than {subschema['minItems']}")

    if "items" in subschema:
        for i, item in enumerate(value):
            errors.extend(_validate_value(f"{path}[{i}]", item, subschema["items"]))

    return errors


def validate_record(record: dict, schema: dict) -> list[str]:
    """Validate a mini-ledger record against the schema; [] means pass.

    The SCHEMA literal below is a verbatim transcription of the accepted
    JSON Schema (only ``false`` -> ``False`` per Python syntax).

    >>> SCHEMA = {
    ...     "$schema": "http://json-schema.org/draft-07/schema#",
    ...     "title": "MiniLedgerRecord",
    ...     "type": "object",
    ...     "properties": {
    ...         "id": {"type": "string", "minLength": 1},
    ...         "amount": {"type": "number", "multipleOf": 0.01},
    ...         "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
    ...         "tags": {"type": "array", "items": {"type": "string"}, "minItems": 0}
    ...     },
    ...     "required": ["id", "amount", "currency", "tags"],
    ...     "additionalProperties": False
    ... }
    >>> validate_record(
    ...     {"id": "TX-001", "amount": 12.34, "currency": "CNY", "tags": []},
    ...     SCHEMA,
    ... )
    []
    >>> validate_record(
    ...     {"id": "TX-002", "amount": 9.999, "currency": "cny1", "tags": [1, "food"]},
    ...     SCHEMA,
    ... )
    ['amount: 9.999 is not a multiple of 0.01', "currency: 'cny1' does not match pattern '^[A-Z]{3}$'", 'tags[0]: expected string, got integer']
    >>> validate_record(
    ...     {"id": "TX-003", "amount": 5.0, "note": "lunch"},
    ...     SCHEMA,
    ... )
    ['currency: required field is missing', 'tags: required field is missing', 'note: additional field is not allowed']
    """
    errors: list[str] = []
    properties = schema.get("properties", {})

    for name in schema.get("required", []):
        if name not in record:
            errors.append(f"{name}: required field is missing")

    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in properties:
                errors.append(f"{key}: additional field is not allowed")

    for name, subschema in properties.items():
        if name in record:
            errors.extend(_validate_value(name, record[name], subschema))

    return errors


if __name__ == "__main__":
    import doctest

    results = doctest.testmod()
    raise SystemExit(1 if results.failed else 0)
```

## §3 确定性说明（doctest 为何稳定）

- **违规顺序固定**：先 `required`（按 required 列表序）→ 再 `additionalProperties`（按 record 键序）→ 再 `properties`（按 schema 键序，数组内按索引）。类型不匹配时跳过该字段其余约束，避免连锁噪音。
- **浮点陷阱已规避**：`multipleOf` 用 `Decimal(str(x))` 精确整除判断，`9.999` 判违、`12.34`/`5.0`/`3` 判过，不受二进制浮点误差影响。
- 运行：`python ledger_validator.py`（退出码 0=通过）或 `python -m doctest ledger_validator.py -v`。
