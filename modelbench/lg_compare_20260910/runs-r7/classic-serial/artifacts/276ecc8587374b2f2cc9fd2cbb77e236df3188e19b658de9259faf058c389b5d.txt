```python
import json
import re

_TYPE_CHECKS = {
    "string": lambda v: isinstance(v, str),
    "number": lambda v: isinstance(v, (int, float)) and not isinstance(v, bool),
    "integer": lambda v: isinstance(v, int) and not isinstance(v, bool),
    "boolean": lambda v: isinstance(v, bool),
    "object": lambda v: isinstance(v, dict),
    "array": lambda v: isinstance(v, list),
    "null": lambda v: v is None,
}


def validate_record(record: dict, schema: dict) -> list[str]:
    """按 schema 校验记账记录，返回违规信息列表（空列表 = 通过）。

    >>> schema = json.loads('''{
    ...   "title": "MiniExpenseRecord",
    ...   "type": "object",
    ...   "properties": {
    ...     "id": {"type": "string"},
    ...     "amount": {"type": "number", "exclusiveMinimum": 0, "multipleOf": 0.01},
    ...     "currency": {"type": "string", "pattern": "^[A-Z]{3}$"},
    ...     "tags": {"type": ["array", "null"], "items": {"type": "string"}}
    ...   },
    ...   "required": ["id", "amount", "currency", "tags"],
    ...   "additionalProperties": false
    ... }''')
    >>> validate_record({"id": "e1", "amount": 12.50, "currency": "CNY", "tags": ["food", "travel"]}, schema)
    []
    >>> validate_record({"id": "e2", "amount": 0, "currency": "CNY", "tags": None}, schema)
    ['amount: must be > 0 (exclusiveMinimum), got 0']
    >>> validate_record({"id": "e3", "amount": 3.141, "currency": "cn", "tags": "food"}, schema)
    ['amount: 3.141 is not a multiple of 0.01', "currency: 'cn' does not match pattern '^[A-Z]{3}$'", 'tags: expected array or null, got str']
    """
    errors: list[str] = []
    props = schema.get("properties", {})
    for key in schema.get("required", []):
        if key not in record:
            errors.append(f"{key}: required field is missing")
    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in props:
                errors.append(f"{key}: not an allowed property")
    for key, sub in props.items():
        if key in record:
            _check_field(key, record[key], sub, errors)
    return errors


def _check_field(name: str, value, sub: dict, errors: list) -> None:
    t = sub.get("type")
    if t is not None:
        allowed = list(t) if isinstance(t, list) else [t]  # 支持联合类型数组
        if not any(_TYPE_CHECKS[a](value) for a in allowed):
            errors.append(
                f"{name}: expected {' or '.join(allowed)}, got {type(value).__name__}"
            )
            return
    if value is None:  # type 含 null 时，null 值通过后不再检查其余子规则
        return
    if "exclusiveMinimum" in sub and isinstance(value, (int, float)) and not isinstance(value, bool):
        if value <= sub["exclusiveMinimum"]:
            errors.append(
                f"{name}: must be > {sub['exclusiveMinimum']} (exclusiveMinimum), got {value}"
            )
    if "multipleOf" in sub and isinstance(value, (int, float)) and not isinstance(value, bool):
        q = value / sub["multipleOf"]
        if abs(q - round(q)) > 1e-9:
            errors.append(f"{name}: {value} is not a multiple of {sub['multipleOf']}")
    if "pattern" in sub and isinstance(value, str) and not re.search(sub["pattern"], value):
        errors.append(f"{name}: {value!r} does not match pattern {sub['pattern']!r}")
    if "items" in sub and isinstance(value, list):
        for i, item in enumerate(value):
            _check_field(f"{name}[{i}]", item, sub["items"], errors)
```
