## 实现（纯标准库，完整无截断）

```python
from decimal import Decimal, InvalidOperation

def validate_record(record: dict, schema: dict) -> list[str]:
    """校验记账记录，返回违规信息列表（空列表 = 通过）。

    >>> schema = {
    ...   "$schema": "https://json-schema.org/draft/2020-12/schema",
    ...   "title": "MiniLedgerRecord",
    ...   "type": "object",
    ...   "properties": {
    ...     "id": { "type": "string", "minLength": 1 },
    ...     "amount": { "type": "number", "multipleOf": 0.01 },
    ...     "currency": { "type": "string", "pattern": "^[A-Z]{3}$" },
    ...     "tags": { "type": ["array", "null"], "items": { "type": "string" } }
    ...   },
    ...   "required": ["id", "amount", "currency"],
    ...   "additionalProperties": False
    ... }
    >>> validate_record({'id': 'r1', 'amount': 12.34, 'currency': 'CNY', 'tags': ['food', None]}, schema)
    ["tags[1]: 期望 string，实际为 NoneType"]
    >>> validate_record({'id': 'r2', 'amount': 0.1, 'currency': 'usd', 'extra': 1}, schema)
    ['currency: 未匹配模式 ^[A-Z]{3}$', 'extra: 不在允许的字段列表中']
    >>> validate_record({'id': '', 'amount': 1.234, 'currency': 'JPY'}, schema)
    ['id: 长度小于 minLength 1', 'amount: 不满足 multipleOf 0.01']
    """
    errors: list[str] = []
    props = schema.get("properties", {})

    if schema.get("additionalProperties") is False:
        for key in record:
            if key not in props:
                errors.append(f"{key}: 不在允许的字段列表中")

    for field in schema.get("required", []):
        if field not in record:
            errors.append(f"{field}: 缺少必填字段")

    for key, spec in props.items():
        if key not in record:
            continue
        value = record[key]
        allowed = spec.get("type")
        allowed_list = allowed if isinstance(allowed, list) else [allowed]
        if not _type_ok(value, allowed_list):
            errors.append(f"{key}: 期望 {'/'.join(allowed_list)}，实际为 {type(value).__name__}")
            continue
        if isinstance(value, str) and "minLength" in spec and len(value) < spec["minLength"]:
            errors.append(f"{key}: 长度小于 minLength {spec['minLength']}")
        if isinstance(value, str) and "pattern" in spec:
            import re
            if not re.search(spec["pattern"], value):
                errors.append(f"{key}: 未匹配模式 {spec['pattern']}")
        if isinstance(value, (int, float)) and not isinstance(value, bool) and "multipleOf" in spec:
            if not _multiple_of(value, spec["multipleOf"]):
                errors.append(f"{key}: 不满足 multipleOf {spec['multipleOf']}")
        if isinstance(value, list) and "items" in spec:
            item_types = spec["items"].get("type")
            item_list = item_types if isinstance(item_types, list) else [item_types]
            for i, item in enumerate(value):
                if not _type_ok(item, item_list):
                    errors.append(f"{key}[{i}]: 期望 {'/'.join(item_list)}，实际为 {type(item).__name__}")

    return errors


def _type_ok(value, allowed: list) -> bool:
    mapping = {"string": str, "number": (int, float), "array": list, "null": type(None), "object": dict, "boolean": bool, "integer": int}
    return any(isinstance(value, mapping.get(t, object)) and not (t == "number" and isinstance(value, bool)) for t in allowed)


def _multiple_of(value, step) -> bool:
    try:
        d = Decimal(str(value)) % Decimal(str(step))
    except InvalidOperation:
        return False
    return d == 0

    doctest.testmod(verbose=True)
```
