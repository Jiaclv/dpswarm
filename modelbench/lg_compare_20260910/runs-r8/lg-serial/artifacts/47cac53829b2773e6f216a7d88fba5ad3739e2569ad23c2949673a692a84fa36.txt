```python
"""validate_record -- 迷你记账记录（MiniLedgerRecord）校验器。

严格依据子任务 A 验收的 JSON Schema（draft 2020-12，
$id: https://example.com/schemas/mini-ledger-record.json）实现。
纯标准库，返回违规信息列表；空列表表示通过。
"""

import re
from typing import Any

# 校验参数全部取自 A 验收 schema（逐字依据）
REQUIRED_FIELDS = ("id", "amount", "currency", "tags")  # "required"
ALLOWED_FIELDS = frozenset(REQUIRED_FIELDS)             # additionalProperties: false
CURRENCY_PATTERN = re.compile(r"^[A-Z]{3}$")            # pattern
ID_MIN_LENGTH = 1                                       # minLength
AMOUNT_MINIMUM = 0                                      # minimum
AMOUNT_MULTIPLE = 0.01                                  # multipleOf
_TOL = 1e-9                                             # 浮点容差


def _is_number(value: Any) -> bool:
    # JSON Schema 的 "number" 不含 bool
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def _check_amount_multiple_of(amount: float) -> bool:
    # 材料附注：用 round(amount / 0.01) 的商规避浮点精度风险
    quotient = round(amount / AMOUNT_MULTIPLE)
    return abs(amount - quotient * AMOUNT_MULTIPLE) < _TOL


def validate_record(record: dict, schema: dict) -> list:
    """校验 record 是否符合 MiniLedgerRecord schema。

    返回违规信息列表；空列表表示通过。

    >>> schema = {"$schema": "https://json-schema.org/draft/2020-12/schema", "$id": "https://example.com/schemas/mini-ledger-record.json", "title": "MiniLedgerRecord", "type": "object", "properties": {"id": {"type": "string", "minLength": 1}, "amount": {"type": "number", "multipleOf": 0.01, "minimum": 0}, "currency": {"type": "string", "pattern": "^[A-Z]{3}$"}, "tags": {"oneOf": [{"type": "null"}, {"type": "array", "items": {"type": "string"}}]}}, "required": ["id", "amount", "currency", "tags"], "additionalProperties": False}
    >>> validate_record({"id": "r-001", "amount": 12.34, "currency": "CNY", "tags": ["food"]}, schema)
    []
    >>> validate_record({"id": "", "amount": -1.005, "currency": "us", "tags": "food", "extra": 1}, schema)
    ['id: must be a string with length >= 1', 'amount: must be >= 0', 'amount: must be a multiple of 0.01 (two decimal places)', 'currency: must match pattern "^[A-Z]{3}$"', 'tags: must be null or an array of strings', 'record: unknown field "extra" (additionalProperties: false)']
    >>> validate_record({"id": "r-002", "amount": 0, "currency": "USD", "tags": None}, schema)
    []
    """
    errors: list = []

    if not isinstance(record, dict):
        return ["record: must be an object"]
    if schema.get("type") != "object":
        return ["record: unsupported schema type"]

    # required
    for field in REQUIRED_FIELDS:
        if field not in record:
            errors.append(f"{field}: missing (required)")

    # additionalProperties: false
    for field in record:
        if field not in ALLOWED_FIELDS:
            errors.append(f'record: unknown field "{field}" (additionalProperties: false)')

    # id: string, minLength: 1
    if "id" in record:
        value = record["id"]
        if not isinstance(value, str) or len(value) < ID_MIN_LENGTH:
            errors.append("id: must be a string with length >= 1")

    # amount: number, multipleOf: 0.01, minimum: 0
    if "amount" in record:
        value = record["amount"]
        if not _is_number(value):
            errors.append("amount: must be a number")
        else:
            if value < AMOUNT_MINIMUM:
                errors.append("amount: must be >= 0")
            if not _check_amount_multiple_of(value):
                errors.append("amount: must be a multiple of 0.01 (two decimal places)")

    # currency: string, pattern ^[A-Z]{3}$
    if "currency" in record:
        value = record["currency"]
        if not isinstance(value, str) or not CURRENCY_PATTERN.match(value):
            errors.append('currency: must match pattern "^[A-Z]{3}$"')

    # tags: oneOf [null, array of strings]
    if "tags" in record:
        value = record["tags"]
        if value is None:
            pass
        elif not isinstance(value, list) or any(not isinstance(item, str) for item in value):
            errors.append("tags: must be null or an array of strings")

    return errors
```
